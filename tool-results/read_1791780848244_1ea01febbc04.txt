   199→describe("nav-structure — route smoke test (DONE WHEN criterion)", () => {
   200→  // Per spec: "every destination URL resolves (this test is the graveyard
   201→  // alarm for future dead links)." We assert every NAV_STRUCTURE route has
   202→  // a matching `case "...": return <...View />` in page.tsx's switch.
   203→  it("every NAV_STRUCTURE route has a matching case in page.tsx's switch", () => {
   204→    const allRoutes = getAllRoutes();
   205→    for (const route of allRoutes) {
   206→      // page.tsx's switch: `case "route": return <SomeView />`
   207→      const casePattern = new RegExp(`case "${route}"\\s*:`);
   208→      expect(
   209→        pageSource,
   210→        `route "${route}" must have a \`case "${route}":\` in page.tsx`,
   211→      ).toMatch(casePattern);
   212→    }
   213→  });
   214→
   215→  it("every NAV_STRUCTURE route resolves to a <...View /> component", () => {
   216→    // page.tsx's switch returns a View component for each case. Three
   217→    // patterns to handle:
   218→    //   1. `case "route": return <SomeView ... />` (direct return)
   219→    //   2. `case "route":\n case "other": return <SomeView ... />` (fall-
   220→    //      through; the `unions` + `my-union` cases share a return)
   221→    //   3. `case "route":\n if (condition) return <SomeView ... />` (guarded
   222→    //      return; the `retention` case has an isRetentionEnabled() guard)
   223→    //
   224→    // The regex allows optional whitespace (including newlines) between
   225→    // `return` and `<`, handles the fall-through by allowing other
   226→    // `case` clauses, and handles the guard by allowing `if (...)` blocks.
   227→    const allRoutes = getAllRoutes();
   228→    for (const route of allRoutes) {
   229→      // Match: `case "route":` followed by (eventually, after comments,
   230→      // fall-through cases, or if-guards) `return <SomeView`.
   231→      // The `s` flag makes `.` match newlines; we use a non-greedy `.*?`
   232→      // so we stop at the FIRST `return <...View` after the case label.
   233→      const casePattern = new RegExp(
   234→        `case "${route}"\\s*:[\\s\\S]*?return\\s*\\(?\\s*<\\w+View`,
   235→      );
   236→      expect(
   237→        pageSource,
   238→        `route "${route}" must return a <...View /> component in page.tsx`,
   239→      ).toMatch(casePattern);
   240→    }
   241→  });
   242→});
