   345→function DocumentTile({
   346→  doc,
   347→  unionId,
   348→  userRole,
   349→  onUploadVersion,
   350→}: {
   351→  doc: UnionDocument;
   352→  unionId: string;
   353→  userRole: string;
   354→  onUploadVersion: () => void;
   355→}) {
   356→  const [versionsOpen, setVersionsOpen] = useState(false);
   357→  const [versions, setVersions] = useState<unknown[] | null>(null);
   358→  const [versionsLoading, setVersionsLoading] = useState(false);
   359→  const [previewDoc, setPreviewDoc] = useState<{
   360→    documentType: string;
   361→    version: number;
   362→    fileName: string;
   363→    fileExtension: string | null;
   364→    downloadUrl: string;
   365→  } | null>(null);
   366→
   367→  const canMutate = userRole !== "INSPECTOR";
   368→
   369→  async function loadVersions() {
   370→    setVersionsLoading(true);
   371→    try {
   372→      const res = await fetch(`/api/unions/${unionId}/documents/${doc.id}/versions`, {
   373→        credentials: "include",
   374→      });
   375→      const data = await res.json();
   376→      if (res.ok) {
   377→        setVersions(data.versions);
   378→      } else {
   379→        toast.error("Failed to load versions.");
   380→      }
   381→    } catch {
   382→      toast.error("Network error.");
   383→    } finally {
   384→      setVersionsLoading(false);
   385→    }
   386→  }
   387→
   388→  async function restoreVersion(versionId: string) {
   389→    try {
   390→      const res = await fetch(`/api/unions/${unionId}/documents/${doc.id}/restore`, {
   391→        method: "POST",
   392→        headers: { "Content-Type": "application/json" },
   393→        credentials: "include",
   394→        body: JSON.stringify({ versionId }),
   395→      });
   396→      const data = await res.json();
   397→      if (!res.ok) {
   398→        toast.error(data.error || "Restore failed.");
   399→        return;
   400→      }
   401→      toast.success(`Restored from version ${data.restoredFromVersion}. New version: ${data.newVersion.versionNumber}.`);
   402→      setVersionsOpen(false);
   403→      // Reload the union list to reflect the new version
   404→      window.location.reload();
   405→    } catch {
   406→      toast.error("Network error.");
   407→    }
   408→  }
   409→
   410→  return (
   411→    <div className="rounded-md border p-3">
   412→      <div className="flex items-start justify-between">
   413→        <div className="flex items-start gap-2">
   414→          <FileText className="mt-0.5 h-4 w-4 text-muted-foreground" />
   415→          <div>
   416→            <p className="text-sm font-medium">{doc.documentType}</p>
   417→            {doc.disposedAt ? (
   418→              <Badge variant="destructive" className="mt-1 text-xs">Disposed</Badge>
   419→            ) : doc.latestVersion ? (
   420→              <div className="mt-1 flex flex-wrap items-center gap-2">
   421→                <p className="text-xs text-muted-foreground">
   422→                  v{doc.latestVersion.versionNumber} · {new Date(doc.latestVersion.uploadedAt).toLocaleDateString()}
   423→                </p>
   424→                {doc.reviewStatus === "PENDING" && (
   425→                  <Badge variant="secondary" className="text-xs bg-amber-100 text-amber-800 border-amber-200">Pending Review</Badge>
   426→                )}
   427→                {doc.reviewStatus === "APPROVED" && (
   428→                  <Badge variant="secondary" className="text-xs bg-green-100 text-green-800 border-green-200">Approved</Badge>
   429→                )}
   430→                {doc.reviewStatus === "REJECTED" && (
   431→                  <Badge variant="secondary" className="text-xs bg-red-100 text-red-800 border-red-200 cursor-help" title={doc.rejectionReason ?? "Rejected by Registrar"}>
   432→                    Rejected
   433→                  </Badge>
   434→                )}
   435→              </div>
   436→            ) : (
   437→              <p className="mt-1 text-xs text-muted-foreground">No version uploaded</p>
   438→            )}
   439→            {doc.reviewStatus === "REJECTED" && doc.rejectionReason && (
   440→              <p className="mt-1 text-xs text-red-600">Reason: {doc.rejectionReason}</p>
   441→            )}
   442→          </div>
   443→        </div>
   444→      </div>
   445→      <div className="mt-3 flex flex-wrap gap-1">
   446→        {canMutate && (
   447→          <Button size="sm" variant="outline" className="h-7 text-xs" onClick={onUploadVersion}>
   448→            <Upload className="mr-1 h-3 w-3" />
   449→            New version
   450→          </Button>
   451→        )}
   452→        {doc.latestVersion && !doc.disposedAt && (
   453→          <>
   454→            <Button
   455→              size="sm"
   456→              variant="ghost"
   457→              className="h-7 text-xs"
   458→              onClick={() =>
   459→                setPreviewDoc({
   460→                  documentType: doc.documentType,
   461→                  version: doc.latestVersion!.versionNumber,
   462→                  fileName: doc.latestVersion!.fileName,
   463→                  fileExtension: doc.latestVersion!.fileExtension,
   464→                  downloadUrl: `/api/unions/${unionId}/documents/${doc.id}/download`,
   465→                })
   466→              }
   467→            >
   468→              <Eye className="mr-1 h-3 w-3" />
   469→              View
   470→            </Button>
   471→            <a href={`/api/unions/${unionId}/documents/${doc.id}/download`} target="_blank" rel="noopener noreferrer">
   472→              <Button size="sm" variant="ghost" className="h-7 text-xs">
   473→                <Download className="mr-1 h-3 w-3" />
   474→                Download
   475→              </Button>
   476→            </a>
   477→          </>
   478→        )}
   479→        <Button
   480→          size="sm"
   481→          variant="ghost"
   482→          className="h-7 text-xs"
   483→          onClick={() => {
   484→            if (!versionsOpen && !versions) loadVersions();
   485→            setVersionsOpen(!versionsOpen);
   486→          }}
   487→        >
   488→          <History className="mr-1 h-3 w-3" />
   489→          History
   490→        </Button>
   491→      </div>
   492→
   493→      {versionsOpen && (
   494→        <div className="mt-3 border-t pt-2">
   495→          {versionsLoading ? (
   496→            <p className="text-xs text-muted-foreground">Loading...</p>
   497→          ) : versions && versions.length > 0 ? (
   498→            <div className="space-y-1">
   499→              {(versions as Array<{
   500→                id: string;
   501→                versionNumber: number;
   502→                fileName: string;
   503→                uploadedAt: string;
   504→                isCurrent: boolean;
   505→              }>).map((v) => (
   506→                <div
   507→                  key={v.id}
   508→                  className="flex items-center justify-between rounded bg-muted/30 p-2 text-xs"
   509→                >
   510→                  <div>
   511→                    <span className="font-medium">v{v.versionNumber}</span>
   512→                    {v.isCurrent && <Badge className="ml-1 text-[10px]">Current</Badge>}
   513→                    <p className="text-muted-foreground">{new Date(v.uploadedAt).toLocaleString()}</p>
   514→                  </div>
   515→                  {!v.isCurrent && canMutate && (
   516→                    <Button
   517→                      size="sm"
   518→                      variant="ghost"
   519→                      className="h-6 text-xs"
   520→                      onClick={() => restoreVersion(v.id)}
   521→                    >
   522→                      <RotateCcw className="mr-1 h-3 w-3" />
   523→                      Restore
   524→                    </Button>
   525→                  )}
   526→                </div>
   527→              ))}
   528→            </div>
   529→          ) : (
   530→            <p className="text-xs text-muted-foreground">No versions.</p>
   531→          )}
   532→        </div>
   533→      )}
   534→
   535→      {previewDoc && (
   536→        <DocumentPreviewModal
   537→          open={!!previewDoc}
   538→          onOpenChange={(open) => !open && setPreviewDoc(null)}
   539→          documentType={previewDoc.documentType}
   540→          version={previewDoc.version}
   541→          fileName={previewDoc.fileName}
   542→          fileExtension={previewDoc.fileExtension}
   543→          downloadUrl={previewDoc.downloadUrl}
   544→        />
   545→      )}
   546→    </div>
   547→  );
   548→}

   554→function EditUnionDialog({
   555→  union,
   556→  userRole,
   557→  onClose,
   558→  onSaved,
   559→}: {
   560→  union: Union;
   561→  userRole: string;
   562→  onClose: () => void;
   563→  onSaved: () => void;
   564→}) {
   565→  const isRegistrar = userRole === "REGISTRAR";
   566→  const isOfficer = userRole === "OFFICER";
   567→
   568→  const [unionName, setUnionName] = useState(union.unionName);
   569→  const [registrationNo, setRegistrationNo] = useState(union.registrationNo);
   570→  const [fileNumber, setFileNumber] = useState(union.fileNumber || "");
   571→  const [status, setStatus] = useState(union.status);
   572→  const [address, setAddress] = useState(union.address || "");
   573→  const [contactPerson, setContactPerson] = useState(union.contactPerson || "");
   574→  const [contactEmail, setContactEmail] = useState(union.contactEmail || "");
   575→  const [contactPhone, setContactPhone] = useState(union.contactPhone || "");
   576→  const [remarks, setRemarks] = useState(union.remarks || "");
   577→  const [saving, setSaving] = useState(false);
   578→
   579→  // Officer restricted fields (silently pinned server-side)
   580→  const restrictedFields = isOfficer
   581→    ? ["unionName", "registrationNo", "fileNumber", "status", "remarks"]

   759→interface UploadDialogProps {
   760→  unionId: string;
   761→  defaultDocType: string;
   762→  open: boolean;
   763→  onClose: () => void;
   764→  onUploaded: () => void;

   767→function UploadDialog({ unionId, defaultDocType, open, onClose, onUploaded }: UploadDialogProps) {
   768→  const [documentType, setDocumentType] = useState(defaultDocType);
   769→  const [title, setTitle] = useState("");
   770→  const [file, setFile] = useState<File | null>(null);
   771→  const [uploading, setUploading] = useState(false);
   772→
   773→  // Sync the doc type when the default changes (e.g., user clicks "New version" on a specific doc)
   774→  useState(() => {
   775→    if (defaultDocType) setDocumentType(defaultDocType);
   776→  });
   777→
   778→  async function handleUpload(e: FormEvent) {
   779→    e.preventDefault();
   780→    if (!file) {
   781→      toast.error("Please select a file.");
   782→      return;
   783→    }
   784→    if (!documentType.trim()) {
   785→      toast.error("Document type is required.");
   786→      return;
   787→    }
   788→    if (!title.trim()) {
   789→      toast.error("Title is required.");
   790→      return;
   791→    }
   792→    setUploading(true);
   793→    try {
   794→      const formData = new FormData();
   795→      formData.append("file", file);
   796→      formData.append("documentType", documentType.trim());
   797→      formData.append("title", title.trim());
   798→      const res = await fetch(`/api/unions/${unionId}/documents`, {
   799→        method: "POST",
   800→        credentials: "include",
   801→        body: formData,
   802→      });
   803→      const data = await res.json();
   804→      if (!res.ok) {
   805→        toast.error(data.error || "Upload failed.");
   806→        return;
   807→      }
   808→      toast.success(`Uploaded ${documentType} v${data.version.versionNumber}.`);
   809→      onUploaded();
   810→    } catch {
   811→      toast.error("Network error. Please try again.");
   812→    } finally {
   813→      setUploading(false);
   814→    }
   815→  }
   816→
   817→  return (
   818→    <Dialog open={open} onOpenChange={(o) => !o && onClose()}>
   819→      <DialogContent>
   820→        <DialogHeader>
   821→          <DialogTitle>Upload Document</DialogTitle>
   822→          <DialogDescription>
   823→            Upload a new version of a document. Duplicate detection is based on SHA-256 checksum.
   824→          </DialogDescription>
   825→        </DialogHeader>
   826→        <form onSubmit={handleUpload} className="space-y-4">
   827→          <div className="space-y-2">
   828→            <Label htmlFor="docType">Document Type</Label>
   829→            <Input
   830→              id="docType"
   831→              value={documentType}
   832→              onChange={(e) => setDocumentType(e.target.value)}
   833→              placeholder="e.g. Constitution, Financial Return"
   834→            />
   835→          </div>
   836→          <div className="space-y-2">
   837→            <Label htmlFor="title">Title</Label>
   838→            <Input
   839→              id="title"
   840→              value={title}
   841→              onChange={(e) => setTitle(e.target.value)}
   842→              placeholder="A descriptive title for this version"
   843→            />
   844→          </div>
   845→          <div className="space-y-2">
   846→            <Label htmlFor="file">File</Label>
   847→            <Input
   848→              id="file"
   849→              type="file"
   850→              onChange={(e) => setFile(e.target.files?.[0] ?? null)}
   851→              accept=".pdf,.doc,.docx,.jpg,.png"
   852→            />
   853→            {file && (
   854→              <p className="text-xs text-muted-foreground">
   855→                {file.name} ({(file.size / 1024).toFixed(1)} KB)
   856→              </p>
   857→            )}
   858→          </div>
   859→          <DialogFooter>
   860→            <Button type="button" variant="outline" onClick={onClose} disabled={uploading}>
   861→              Cancel
   862→            </Button>
   863→            <Button type="submit" disabled={uploading || !file}>
   864→              {uploading ? (
   865→                <>
   866→                  <Loader2 className="mr-2 h-4 w-4 animate-spin" />
   867→                  Uploading...
   868→                </>
   869→              ) : (
   870→                <>
   871→                  <Upload className="mr-2 h-4 w-4" />
   872→                  Upload
   873→                </>
   874→              )}
   875→            </Button>
   876→          </DialogFooter>
   877→        </form>
   878→      </DialogContent>
   879→    </Dialog>
   880→  );
