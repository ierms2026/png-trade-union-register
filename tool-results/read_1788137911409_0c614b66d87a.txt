    96→  const selectedUnion = data?.unions.find((u) => u.id === selectedUnionId);

    99→  // network round-trip. Filters by union name or registration number.

   105→      (u) =>
   106→        u.unionName.toLowerCase().includes(q) ||
   107→        u.registrationNo.toLowerCase().includes(q),
   108→    );

   111→  if (loading) return <CardListSkeleton count={6} />;

   142→                </CardDescription>

   146→                  size="sm"

   148→                  onClick={() => setEditDialogOpen(true)}

   150→                  <Pencil className="mr-2 h-4 w-4" />

   170→                  <p className="text-xs text-muted-foreground">File Number</p>

   315→// -----------------------------------------------------------------------------

   318→
   319→function DocumentTile({
   320→  doc,
   321→  unionId,
   322→  onUploadVersion,
   323→}: {
   324→  doc: UnionDocument;
   325→  unionId: string;
   326→  onUploadVersion: () => void;
   327→}) {
   328→  const [versionsOpen, setVersionsOpen] = useState(false);
   329→  const [versions, setVersions] = useState<unknown[] | null>(null);
   330→  const [versionsLoading, setVersionsLoading] = useState(false);
   331→
   332→  async function loadVersions() {
   333→    setVersionsLoading(true);
   334→    try {
   335→      const res = await fetch(`/api/unions/${unionId}/documents/${doc.id}/versions`, {
   336→        credentials: "include",
   337→      });
   338→      const data = await res.json();
   339→      if (res.ok) {
   340→        setVersions(data.versions);
   341→      } else {
   342→        toast.error("Failed to load versions.");
   343→      }
   344→    } catch {
   345→      toast.error("Network error.");
   346→    } finally {
   347→      setVersionsLoading(false);
   348→    }
   349→  }
   350→
   351→  async function restoreVersion(versionId: string) {
   352→    try {
   353→      const res = await fetch(`/api/unions/${unionId}/documents/${doc.id}/restore`, {
   354→        method: "POST",
   355→        headers: { "Content-Type": "application/json" },
   356→        credentials: "include",
   357→        body: JSON.stringify({ versionId }),
   358→      });
   359→      const data = await res.json();
   360→      if (!res.ok) {
   361→        toast.error(data.error || "Restore failed.");
   362→        return;
   363→      }
   364→      toast.success(`Restored from version ${data.restoredFromVersion}. New version: ${data.newVersion.versionNumber}.`);
   365→      setVersionsOpen(false);
   366→      // Reload the union list to reflect the new version
   367→      window.location.reload();
   368→    } catch {
   369→      toast.error("Network error.");
   370→    }
   371→  }
   372→
   373→  return (
   374→    <div className="rounded-md border p-3">
   375→      <div className="flex items-start justify-between">
   376→        <div className="flex items-start gap-2">
   377→          <FileText className="mt-0.5 h-4 w-4 text-muted-foreground" />
   378→          <div>
   379→            <p className="text-sm font-medium">{doc.documentType}</p>
   380→            {doc.disposedAt ? (
   381→              <Badge variant="destructive" className="mt-1 text-xs">Disposed</Badge>
   382→            ) : doc.latestVersion ? (
   383→              <p className="mt-1 text-xs text-muted-foreground">
   384→                v{doc.latestVersion.versionNumber} · {new Date(doc.latestVersion.uploadedAt).toLocaleDateString()}
   385→              </p>
   386→            ) : (
   387→              <p className="mt-1 text-xs text-muted-foreground">No version uploaded</p>
   388→            )}
   389→          </div>
   390→        </div>
   391→      </div>
   392→      <div className="mt-3 flex gap-1">
   393→        <Button size="sm" variant="outline" className="h-7 text-xs" onClick={onUploadVersion}>
   394→          <Upload className="mr-1 h-3 w-3" />
   395→          New version
   396→        </Button>
   397→        {doc.latestVersion && !doc.disposedAt && (
   398→          <a href={`/api/unions/${unionId}/documents/${doc.id}/download`} target="_blank" rel="noopener noreferrer">
   399→            <Button size="sm" variant="ghost" className="h-7 text-xs">
   400→              <Download className="mr-1 h-3 w-3" />
   401→              Download
   402→            </Button>
   403→          </a>
   404→        )}
   405→        <Button
   406→          size="sm"
   407→          variant="ghost"
   408→          className="h-7 text-xs"
   409→          onClick={() => {
   410→            if (!versionsOpen && !versions) loadVersions();
   411→            setVersionsOpen(!versionsOpen);
   412→          }}
   413→        >
   414→          <History className="mr-1 h-3 w-3" />
   415→          History
   416→        </Button>
   417→      </div>
   418→
   419→      {versionsOpen && (
   420→        <div className="mt-3 border-t pt-2">
   421→          {versionsLoading ? (
   422→            <p className="text-xs text-muted-foreground">Loading...</p>
   423→          ) : versions && versions.length > 0 ? (
   424→            <div className="space-y-1">
   425→              {(versions as Array<{
   426→                id: string;
   427→                versionNumber: number;
   428→                fileName: string;
   429→                uploadedAt: string;
   430→                isCurrent: boolean;
   431→              }>).map((v) => (
   432→                <div
   433→                  key={v.id}
   434→                  className="flex items-center justify-between rounded bg-muted/30 p-2 text-xs"
   435→                >
   436→                  <div>
   437→                    <span className="font-medium">v{v.versionNumber}</span>
   438→                    {v.isCurrent && <Badge className="ml-1 text-[10px]">Current</Badge>}
   439→                    <p className="text-muted-foreground">{new Date(v.uploadedAt).toLocaleString()}</p>
   440→                  </div>
   441→                  {!v.isCurrent && (
   442→                    <Button
   443→                      size="sm"
   444→                      variant="ghost"
   445→                      className="h-6 text-xs"
   446→                      onClick={() => restoreVersion(v.id)}
   447→                    >
   448→                      <RotateCcw className="mr-1 h-3 w-3" />
   449→                      Restore
   450→                    </Button>
   451→                  )}
   452→                </div>
   453→              ))}
   454→            </div>
   455→          ) : (
   456→            <p className="text-xs text-muted-foreground">No versions.</p>
   457→          )}
   458→        </div>
   459→      )}
   460→    </div>
   461→  );
   462→}

   468→function EditUnionDialog({
   469→  union,
   470→  userRole,
   471→  onClose,
   472→  onSaved,
   473→}: {
   474→  union: Union;
   475→  userRole: string;
   476→  onClose: () => void;
   477→  onSaved: () => void;
   478→}) {
   479→  const isRegistrar = userRole === "REGISTRAR";
   480→  const isOfficer = userRole === "OFFICER";
   481→
   482→  const [unionName, setUnionName] = useState(union.unionName);
   483→  const [registrationNo, setRegistrationNo] = useState(union.registrationNo);
   484→  const [fileNumber, setFileNumber] = useState(union.fileNumber || "");
   485→  const [status, setStatus] = useState(union.status);
   486→  const [address, setAddress] = useState(union.address || "");
   487→  const [contactPerson, setContactPerson] = useState(union.contactPerson || "");
   488→  const [contactEmail, setContactEmail] = useState(union.contactEmail || "");
   489→  const [contactPhone, setContactPhone] = useState(union.contactPhone || "");
   490→  const [remarks, setRemarks] = useState(union.remarks || "");
   491→  const [saving, setSaving] = useState(false);
   492→
   493→  // Officer restricted fields (silently pinned server-side)
   494→  const restrictedFields = isOfficer
   495→    ? ["unionName", "registrationNo", "fileNumber", "status", "remarks"]
   496→    : [];
   497→
   498→  async function handleSave(e: FormEvent) {
   499→    e.preventDefault();
   500→    setSaving(true);
   501→    try {
   502→      const body: Record<string, unknown> = {};
   503→      if (isRegistrar) {
   504→        body.unionName = unionName;
   505→        body.registrationNo = registrationNo;
   506→        body.fileNumber = fileNumber || null;
   507→        body.status = status;
   508→        body.remarks = remarks || null;
   509→      }
   510→      body.address = address || null;
   511→      body.contactPerson = contactPerson || null;
   512→      body.contactEmail = contactEmail || null;
   513→      body.contactPhone = contactPhone || null;
   514→
   515→      const res = await fetch(`/api/unions/${union.id}`, {
   516→        method: "PUT",
   517→        headers: { "Content-Type": "application/json" },
   518→        credentials: "include",
   519→        body: JSON.stringify(body),
   520→      });
   521→      const data = await res.json();
   522→      if (!res.ok) {
   523→        toast.error(data.error || "Failed to update union.");
   524→        return;
   525→      }
   526→      toast.success("Union updated successfully.");
   527→      onSaved();
   528→    } catch {
   529→      toast.error("Network error.");
   530→    } finally {
   531→      setSaving(false);
   532→    }
   533→  }
   534→
   535→  return (
   536→    <Dialog open onOpenChange={onClose}>
   537→      <DialogContent className="max-w-lg">
   538→        <DialogHeader>
   539→          <DialogTitle>Edit Union</DialogTitle>
   540→          <DialogDescription>
   541→            {isOfficer
   542→              ? "You can edit contact information only. Core identity fields are protected."
   543→              : "Update union details. Changes are logged in the audit trail."}
   544→          </DialogDescription>
   545→        </DialogHeader>
   546→        <form onSubmit={handleSave} className="space-y-4">
   547→          {/* Union Name — Registrar only */}
   548→          <div className="space-y-2">
   549→            <Label>Union Name</Label>
   550→            <Input
   551→              value={unionName}
   552→              onChange={(e) => setUnionName(e.target.value)}
   553→              disabled={isOfficer}
   554→              className={isOfficer ? "bg-muted" : ""}
   555→            />
   556→            {isOfficer && (
   557→              <p className="text-xs text-muted-foreground">Restricted — only Registrars can change this.</p>
   558→            )}
   559→          </div>
   560→
   561→          {/* Registration No — Registrar only */}
   562→          <div className="grid grid-cols-2 gap-4">
   563→            <div className="space-y-2">
   564→              <Label>Registration No</Label>
   565→              <Input
   566→                value={registrationNo}
   567→                onChange={(e) => setRegistrationNo(e.target.value)}
   568→                disabled={isOfficer}
   569→                className={isOfficer ? "bg-muted" : ""}
   570→              />
   571→            </div>
   572→            <div className="space-y-2">
   573→              <Label>File Number</Label>
   574→              <Input
   575→                value={fileNumber}
   576→                onChange={(e) => setFileNumber(e.target.value)}
   577→                disabled={isOfficer}
   578→                className={isOfficer ? "bg-muted" : ""}
   579→                placeholder="Optional"
   580→              />
   581→            </div>
   582→          </div>
   583→
   584→          {/* Status — Registrar only */}
   585→          {isRegistrar && (
   586→            <div className="space-y-2">
   587→              <Label>Status</Label>
   588→              <Select value={status} onValueChange={setStatus}>

   673→interface UploadDialogProps {
   674→  unionId: string;
   675→  defaultDocType: string;
   676→  open: boolean;
   677→  onClose: () => void;
   678→  onUploaded: () => void;
   679→}
   680→
   681→function UploadDialog({ unionId, defaultDocType, open, onClose, onUploaded }: UploadDialogProps) {
   682→  const [documentType, setDocumentType] = useState(defaultDocType);
   683→  const [title, setTitle] = useState("");
   684→  const [file, setFile] = useState<File | null>(null);
   685→  const [uploading, setUploading] = useState(false);
   686→
   687→  // Sync the doc type when the default changes (e.g., user clicks "New version" on a specific doc)
   688→  useState(() => {
   689→    if (defaultDocType) setDocumentType(defaultDocType);
   690→  });
   691→
   692→  async function handleUpload(e: FormEvent) {
   693→    e.preventDefault();
   694→    if (!file) {
   695→      toast.error("Please select a file.");
   696→      return;
   697→    }
   698→    if (!documentType.trim()) {
   699→      toast.error("Document type is required.");
   700→      return;
   701→    }
   702→    if (!title.trim()) {
   703→      toast.error("Title is required.");
   704→      return;
   705→    }
   706→    setUploading(true);
   707→    try {
   708→      const formData = new FormData();
   709→      formData.append("file", file);
   710→      formData.append("documentType", documentType.trim());
   711→      formData.append("title", title.trim());
   712→      const res = await fetch(`/api/unions/${unionId}/documents`, {
   713→        method: "POST",
   714→        credentials: "include",
   715→        body: formData,
   716→      });
   717→      const data = await res.json();
   718→      if (!res.ok) {
   719→        toast.error(data.error || "Upload failed.");
   720→        return;
   721→      }
   722→      toast.success(`Uploaded ${documentType} v${data.version.versionNumber}.`);
   723→      onUploaded();
   724→    } catch {
   725→      toast.error("Network error. Please try again.");
   726→    } finally {
   727→      setUploading(false);
   728→    }
   729→  }
   730→
   731→  return (
   732→    <Dialog open={open} onOpenChange={(o) => !o && onClose()}>
   733→      <DialogContent>
   734→        <DialogHeader>
   735→          <DialogTitle>Upload Document</DialogTitle>
   736→          <DialogDescription>
   737→            Upload a new version of a document. Duplicate detection is based on SHA-256 checksum.
   738→          </DialogDescription>
   739→        </DialogHeader>
   740→        <form onSubmit={handleUpload} className="space-y-4">
   741→          <div className="space-y-2">
   742→            <Label htmlFor="docType">Document Type</Label>
   743→            <Input
   744→              id="docType"
   745→              value={documentType}
   746→              onChange={(e) => setDocumentType(e.target.value)}
   747→              placeholder="e.g. Constitution, Financial Return"
   748→            />
   749→          </div>
   750→          <div className="space-y-2">
   751→            <Label htmlFor="title">Title</Label>
   752→            <Input
   753→              id="title"
   754→              value={title}
   755→              onChange={(e) => setTitle(e.target.value)}
   756→              placeholder="A descriptive title for this version"
   757→            />
   758→          </div>
   759→          <div className="space-y-2">
   760→            <Label htmlFor="file">File</Label>
   761→            <Input
   762→              id="file"
   763→              type="file"
   764→              onChange={(e) => setFile(e.target.files?.[0] ?? null)}
   765→              accept=".pdf,.doc,.docx,.jpg,.png"
   766→            />
   767→            {file && (
   768→              <p className="text-xs text-muted-foreground">
   769→                {file.name} ({(file.size / 1024).toFixed(1)} KB)
   770→              </p>
   771→            )}
   772→          </div>
   773→          <DialogFooter>
   774→            <Button type="button" variant="outline" onClick={onClose} disabled={uploading}>
   775→              Cancel
   776→            </Button>
   777→            <Button type="submit" disabled={uploading || !file}>
   778→              {uploading ? (
   779→                <>
   780→                  <Loader2 className="mr-2 h-4 w-4 animate-spin" />
   781→                  Uploading...
   782→                </>
   783→              ) : (
   784→                <>
   785→                  <Upload className="mr-2 h-4 w-4" />
   786→                  Upload
   787→                </>
   788→              )}
   789→            </Button>
   790→          </DialogFooter>
   791→        </form>
   792→      </DialogContent>
   793→    </Dialog>
   794→  );
   795→}
