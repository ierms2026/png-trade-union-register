   117→      (u) =>
   118→        u.unionName.toLowerCase().includes(q) ||

   126→    return (

   128→        icon={Building2}
   129→        title="No unions registered"

   134→

   159→                  target="_blank"

   161→                >
   162→                  <Button size="sm" variant="ghost">
   163→                    <Globe className="mr-2 h-4 w-4" />
   164→                    View public record
   165→                  </Button>
   166→                </Link>
   167→                <Button
   168→                  size="sm"
   169→                  variant="outline"
   170→                  onClick={() => setEditDialogOpen(true)}
   171→                >
   172→                  <Pencil className="mr-2 h-4 w-4" />

   177→                  onClick={() => {
   178→                    setUploadDialogOpen(true);
   179→                  }}

   252→
   253→        {uploadDialogOpen && (

   270→        {editDialogOpen && (

   283→  }
   284→
   285→  // --- Union list view ---
   286→  return (

   288→      <div className="flex items-center justify-between">

   290→          <h1 className="text-3xl font-bold tracking-tight">Registered Unions</h1>

   292→            {data.unions.length} registered union{data.unions.length !== 1 ? "s" : ""}.

   332→            </div>

   335→                key={u.id}

   353→      {registerDialogOpen && (
   354→        <RegisterUnionDialog

   370→function DocumentTile({
   371→  doc,
   372→  unionId,
   373→  userRole,
   374→  onUploadVersion,
   375→}: {
   376→  doc: UnionDocument;
   377→  unionId: string;
   378→  userRole: string;
   379→  onUploadVersion: () => void;
   380→}) {
   381→  const [versionsOpen, setVersionsOpen] = useState(false);
   382→  const [versions, setVersions] = useState<unknown[] | null>(null);
   383→  const [versionsLoading, setVersionsLoading] = useState(false);
   384→  const [previewDoc, setPreviewDoc] = useState<{
   385→    documentType: string;
   386→    version: number;
   387→    fileName: string;
   388→    fileExtension: string | null;
   389→    downloadUrl: string;
   390→  } | null>(null);
   391→
   392→  const canMutate = userRole !== "INSPECTOR";
   393→
   394→  async function loadVersions() {
   395→    setVersionsLoading(true);
   396→    try {
   397→      const res = await fetch(`/api/unions/${unionId}/documents/${doc.id}/versions`, {
   398→        credentials: "include",
   399→      });
   400→      const data = await res.json();
   401→      if (res.ok) {
   402→        setVersions(data.versions);
   403→      } else {
   404→        toast.error("Failed to load versions.");
   405→      }
   406→    } catch {
   407→      toast.error("Network error.");
   408→    } finally {
   409→      setVersionsLoading(false);
   410→    }
   411→  }
   412→
   413→  async function restoreVersion(versionId: string) {
   414→    try {
   415→      const res = await fetch(`/api/unions/${unionId}/documents/${doc.id}/restore`, {
   416→        method: "POST",
   417→        headers: { "Content-Type": "application/json" },
   418→        credentials: "include",
   419→        body: JSON.stringify({ versionId }),
   420→      });
   421→      const data = await res.json();
   422→      if (!res.ok) {
   423→        toast.error(data.error || "Restore failed.");
   424→        return;
   425→      }
   426→      toast.success(`Restored from version ${data.restoredFromVersion}. New version: ${data.newVersion.versionNumber}.`);
   427→      setVersionsOpen(false);
   428→      // Reload the union list to reflect the new version
   429→      window.location.reload();
   430→    } catch {
   431→      toast.error("Network error.");
   432→    }
   433→  }
   434→
   435→  return (
   436→    <div className="rounded-md border p-3">
   437→      <div className="flex items-start justify-between">
   438→        <div className="flex items-start gap-2">
   439→          <FileText className="mt-0.5 h-4 w-4 text-muted-foreground" />
   440→          <div>
   441→            <p className="text-sm font-medium">
   442→              {doc.documentType}
   443→              {doc.isPublic && (
   444→                <Badge variant="outline" className="ml-2 text-[10px] text-green-600 border-green-600/30">
   445→                  <Globe className="mr-0.5 h-2.5 w-2.5" />
   446→                  Public
   447→                </Badge>
   448→              )}
   449→            </p>
   450→            {doc.disposedAt ? (
   451→              <Badge variant="destructive" className="mt-1 text-xs">Disposed</Badge>
   452→            ) : doc.latestVersion ? (
   453→              <div className="mt-1 flex flex-wrap items-center gap-2">
   454→                <p className="text-xs text-muted-foreground">
   455→                  v{doc.latestVersion.versionNumber} · {new Date(doc.latestVersion.uploadedAt).toLocaleDateString()}
   456→                </p>
   457→                {doc.reviewStatus === "PENDING" && (
   458→                  <Badge variant="secondary" className="text-xs bg-amber-100 text-amber-800 border-amber-200">Pending Review</Badge>
   459→                )}
   460→                {doc.reviewStatus === "APPROVED" && (
   461→                  <Badge variant="secondary" className="text-xs bg-green-100 text-green-800 border-green-200">Approved</Badge>
   462→                )}
   463→                {doc.reviewStatus === "REJECTED" && (
   464→                  <Badge variant="secondary" className="text-xs bg-red-100 text-red-800 border-red-200 cursor-help" title={doc.rejectionReason ?? "Rejected by Registrar"}>
   465→                    Rejected
   466→                  </Badge>
   467→                )}
   468→              </div>
   469→            ) : (
   470→              <p className="mt-1 text-xs text-muted-foreground">No version uploaded</p>
   471→            )}
   472→            {doc.reviewStatus === "REJECTED" && doc.rejectionReason && (
   473→              <p className="mt-1 text-xs text-red-600">Reason: {doc.rejectionReason}</p>
   474→            )}
   475→          </div>
   476→        </div>
   477→      </div>
   478→      <div className="mt-3 flex flex-wrap gap-1">
   479→        {canMutate && (
   480→          <Button size="sm" variant="outline" className="h-7 text-xs" onClick={onUploadVersion}>
   481→            <Upload className="mr-1 h-3 w-3" />
   482→            New version
   483→          </Button>
   484→        )}
   485→        {doc.latestVersion && !doc.disposedAt && (
   486→          <>
   487→            <Button
   488→              size="sm"
   489→              variant="ghost"
   490→              className="h-7 text-xs"
   491→              onClick={() =>
   492→                setPreviewDoc({
   493→                  documentType: doc.documentType,
   494→                  version: doc.latestVersion!.versionNumber,
   495→                  fileName: doc.latestVersion!.fileName,
   496→                  fileExtension: doc.latestVersion!.fileExtension,
   497→                  downloadUrl: `/api/unions/${unionId}/documents/${doc.id}/download`,
   498→                })
   499→              }
   500→            >
   501→              <Eye className="mr-1 h-3 w-3" />
   502→              View
   503→            </Button>
   504→            <a href={`/api/unions/${unionId}/documents/${doc.id}/download`} target="_blank" rel="noopener noreferrer">
   505→              <Button size="sm" variant="ghost" className="h-7 text-xs">
   506→                <Download className="mr-1 h-3 w-3" />
   507→                Download
   508→              </Button>
   509→            </a>
   510→          </>
   511→        )}
   512→        <Button
   513→          size="sm"
   514→          variant="ghost"
   515→          className="h-7 text-xs"
   516→          onClick={() => {
   517→            if (!versionsOpen && !versions) loadVersions();
   518→            setVersionsOpen(!versionsOpen);
   519→          }}
   520→        >
   521→          <History className="mr-1 h-3 w-3" />
   522→          History
   523→        </Button>
   524→        {/* Registrar-only: Publish/Unpublish toggle for public portal visibility */}
   525→        {userRole === "REGISTRAR" && !doc.disposedAt && (
   526→          <Button
   527→            size="sm"
   528→            variant="ghost"
   529→            className={`h-7 text-xs ${doc.isPublic ? "text-green-600" : "text-muted-foreground"}`}
   530→            onClick={async () => {
   531→              try {
   532→                const res = await fetch(
   533→                  `/api/unions/${unionId}/documents/${doc.id}/visibility`,
   534→                  {
   535→                    method: "PATCH",
   536→                    headers: { "Content-Type": "application/json" },
   537→                    credentials: "include",
   538→                    body: JSON.stringify({ isPublic: !doc.isPublic }),
   539→                  },
   540→                );
   541→                if (!res.ok) {
   542→                  const err = await res.json();
   543→                  toast.error(err.error || "Failed to update visibility.");
   544→                  return;
   545→                }
   546→                toast.success(
   547→                  doc.isPublic
   548→                    ? `${doc.documentType} unpublished from public portal.`
   549→                    : `${doc.documentType} published to public portal.`,
   550→                );
   551→                // Trigger a refetch by reloading the page data
   552→                window.location.reload();
   553→              } catch {
   554→                toast.error("Network error.");
   555→              }
   556→            }}
   557→          >
   558→            {doc.isPublic ? (
   559→              <>
   560→                <EyeOff className="mr-1 h-3 w-3" />
   561→                Unpublish
   562→              </>
   563→            ) : (
   564→              <>
   565→                <Globe className="mr-1 h-3 w-3" />
   566→                Publish
   567→              </>
   568→            )}
   569→          </Button>
   570→        )}
   571→      </div>
   572→
   573→      {versionsOpen && (
   574→        <div className="mt-3 border-t pt-2">
   575→          {versionsLoading ? (
   576→            <p className="text-xs text-muted-foreground">Loading...</p>
   577→          ) : versions && versions.length > 0 ? (
   578→            <div className="space-y-1">
   579→              {(versions as Array<{
   580→                id: string;
   581→                versionNumber: number;
   582→                fileName: string;
   583→                uploadedAt: string;
   584→                isCurrent: boolean;
   585→              }>).map((v) => (
   586→                <div

   847→function UploadDialog({ unionId, defaultDocType, open, onClose, onUploaded }: UploadDialogProps) {

   855→  const documentTypes = complianceData?.documentTypes ?? [];

   858→  useEffect(() => {
   859→    if (defaultDocType) setDocumentType(defaultDocType);
   860→  }, [defaultDocType]);
   861→
   862→  async function handleUpload(e: FormEvent) {
   863→    e.preventDefault();
   864→    if (!file) {
   865→      toast.error("Please select a file.");
   866→      return;
   867→    }

   873→      toast.error("Title is required.");
   874→      return;

   878→      const formData = new FormData();
   879→      formData.append("file", file);
   880→      formData.append("documentType", documentType.trim());
   881→      formData.append("title", title.trim());

   883→        method: "POST",
   884→        credentials: "include",
   885→        body: formData,
   886→      });

   889→        toast.error(data.error || "Upload failed.");
   890→        return;
   891→      }
   892→      toast.success(`Uploaded ${documentType} v${data.version.versionNumber}.`);
   893→      onUploaded();

   895→      toast.error("Network error. Please try again.");
