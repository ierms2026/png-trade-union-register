   395→/**
   396→ * Build a mailto: URL for the "Give feedback" link, pre-filled with the
   397→ * current page URL and the user's role. The body is a short prompt so
   398→ * the user knows what to type.
   399→ *
   400→ * @param currentUrl The current page URL (window.location.href from the caller).
   401→ * @param roleLabel The user's role label (e.g. "Registrar", "Officer").
   402→ * @returns A mailto: URL string.
   403→ */
   404→export function buildFeedbackMailto(currentUrl: string, roleLabel: string): string {
   405→  const subject = `Feedback on the Register of Trade Unions`;
   406→  const body =
   407→    `I'd like to give feedback on the Register of Trade Unions.\n\n` +
   408→    `Page: ${currentUrl}\n` +
   409→    `My role: ${roleLabel}\n\n` +
   410→    `What I was trying to do:\n\n` +
   411→    `What I expected:\n\n` +
   412→    `What actually happened:\n\n` +
   413→    `(Delete anything above that isn't relevant. The more specific you are, ` +
   414→    `the easier it is to act on.)`;
   415→  // encodeURI handles the common cases. Line breaks (%0A) need explicit
   416→  // handling because encodeURI does NOT encode them.
   417→  const encoded = `mailto:${FEEDBACK_EMAIL}?subject=${encodeURIComponent(subject)}&body=${encodeURIComponent(body)}`;
   418→  return encoded;
   419→}

   505→/**
   506→ * Group a flat list of document type strings into categories.
   507→ * Types not in any category go into an "Other" group at the bottom.
   508→ * Categories with zero matching types are omitted.
   509→ */
   510→export function groupDocumentTypesByCategory(
   511→  documentTypes: string[],
   512→): DocumentTypeCategory[] {
   513→  const typeSet = new Set(documentTypes);
   514→  const grouped: DocumentTypeCategory[] = [];
   515→  const seen = new Set<string>();
   516→
   517→  for (const cat of DOCUMENT_TYPE_CATEGORIES) {
   518→    const inCategory = cat.documentTypes.filter((t) => typeSet.has(t));
   519→    if (inCategory.length > 0) {
   520→      grouped.push({ label: cat.label, description: cat.description, documentTypes: inCategory });
   521→      for (const t of inCategory) seen.add(t);
   522→    }
   523→  }
   524→
   525→  // "Other" group for types not in any category
   526→  const others = documentTypes.filter((t) => !seen.has(t));
   527→  if (others.length > 0) {
   528→    grouped.push({ label: "Other", description: "Document types not in a standard category.", documentTypes: others });
   529→  }
   530→
   531→  return grouped;
   532→}

   605→// =============================================================================
   606→// Notification linkUrl helpers — canonical format for all notification links.
   607→//
   608→// CANONICAL FORMAT (one agreed style, used everywhere):
   609→//   Staff notifications:  /?view=<view>       (e.g., /?view=reviews)
   610→//   Portal notifications: /portal?view=<view> (e.g., /portal?view=contact)
   611→//
   612→// Every notification creator calls staffLinkUrl() or portalLinkUrl() — never
   613→// string-builds. Every consumer (AppShell, and historically PortalShell
   614→// before the /portal app was removed in D7) calls the matching
   615→// audience-aware parser. Email templates call resolveLinkUrlForEmail() to
   616→// get an absolute URL that works when clicked from an email client.
   617→//
   618→// AUDIENCE-AWARE PARSING (prevents silent cross-audience failures):
   619→//   parseStaffLinkUrl  — only honors /?view=X, IGNORES /portal?view=X.
   620→//   parsePortalLinkUrl — only honors /portal?view=X, IGNORES /?view=X.
   621→//
   622→// Why: before audience-aware parsing, a staff user who received a portal
   623→// link (/portal?view=contact) would have "contact" extracted by the generic
   624→// parseLinkUrl, then onNavigate("contact") would fire — but the staff app
   625→// has no "contact" view, so the user silently landed on the Dashboard.
   626→// With audience-aware parsing, the staff app ignores the portal link entirely
   627→// (the notification is still marked as read, just no navigation).
   628→//
   629→// The generic parseLinkUrl() is kept for backward compat + debugging — it
   630→// extracts ?view= from any link regardless of audience. Production code
   631→// should use the audience-aware variants.
   632→// =============================================================================
   633→
   634→/**
   635→ * Build a canonical staff notification linkUrl.
   636→ * @param view The view key (e.g., "reviews", "compliance", "tasks")
   637→ * @returns A URL like "/?view=reviews"
   638→ */
   639→export function staffLinkUrl(view: string): string {
   640→  return `/?view=${view}`;
   641→}
   642→
   643→/**
   644→ * Build a canonical portal notification linkUrl.
   645→ * @param view The portal view key (e.g., "contact", "documents", "compliance")
   646→ * @returns A URL like "/portal?view=contact"
   647→ */
   648→export function portalLinkUrl(view: string): string {
   649→  return `/portal?view=${view}`;
   650→}
   651→
   652→/**
   653→ * Parse a notification linkUrl + extract the view key.
   654→ * Works for both staff (/?view=X) and portal (/portal?view=X) formats.
   655→ *
   656→ * This is the GENERIC parser — it extracts ?view= from any link regardless
   657→ * of audience. Production code should use the audience-aware variants
   658→ * (parseStaffLinkUrl, parsePortalLinkUrl) to prevent cross-audience silent
   659→ * failures. This generic parser is kept for backward compat + debugging.
   660→ *
   661→ * @returns The view key (e.g., "reviews"), or null if no ?view= param found.
   662→ */
   663→export function parseLinkUrl(linkUrl: string | null | undefined): string | null {
   664→  if (!linkUrl) return null;
   665→  const match = linkUrl.match(/[?&]view=([^&]+)/);
   666→  return match ? decodeURIComponent(match[1]) : null;
   667→}
   668→
   669→/**
   670→ * Parse a STAFF notification linkUrl — only honors /?view=X format.
   671→ * Ignores portal links (/portal?view=X) to prevent cross-audience silent
   672→ * failures. Used by AppShell (the staff notification bell).
   673→ *
   674→ * @returns The view key (e.g., "reviews"), or null if the link is not a
   675→ *          staff link or has no ?view= param.
   676→ */
   677→export function parseStaffLinkUrl(linkUrl: string | null | undefined): string | null {
   678→  if (!linkUrl) return null;
   679→  // Staff links start with /?view= or just ?view= (relative). They do NOT
   680→  // start with /portal. This prevents a portal link from being honored by
   681→  // the staff app.
   682→  if (linkUrl.startsWith("/portal")) return null;
   683→  const match = linkUrl.match(/[?&]view=([^&]+)/);
   684→  return match ? decodeURIComponent(match[1]) : null;
   685→}
   686→
   687→/**
   688→ * Parse a PORTAL notification linkUrl — only honors /portal?view=X format.
   689→ * Ignores staff links (/?view=X) to prevent cross-audience silent failures.
   690→ * Used historically by PortalShell (the representative notification bell,
   691→ * removed in D7). Kept because /api/contact-requests/[id]/{approve,reject}
   692→ * still calls portalLinkUrl() to build notification linkUrls.
   693→ *
   694→ * @returns The view key (e.g., "contact"), or null if the link is not a
   695→ *          portal link or has no ?view= param.
   696→ */
   697→export function parsePortalLinkUrl(linkUrl: string | null | undefined): string | null {
   698→  if (!linkUrl) return null;
   699→  // Portal links must start with /portal. This prevents a staff link
   700→  // (/?view=X) from being honored by the portal.
   701→  if (!linkUrl.startsWith("/portal")) return null;
   702→  const match = linkUrl.match(/[?&]view=([^&]+)/);
   703→  return match ? decodeURIComponent(match[1]) : null;
   704→}
   705→
   706→/**
   707→ * Resolve a relative linkUrl to an absolute URL for email rendering.
   708→ * Uses NEXT_PUBLIC_APP_URL if set, otherwise falls back to the provided base URL.
   709→ *
   710→ * @param linkUrl The relative linkUrl (e.g., "/?view=reviews")
   711→ * @param fallbackBaseUrl The base URL to use if NEXT_PUBLIC_APP_URL is not set
   712→ * @returns An absolute URL (e.g., "http://localhost:3000/?view=reviews")
   713→ */
   714→export function resolveLinkUrlForEmail(
   715→  linkUrl: string | null | undefined,
   716→  fallbackBaseUrl: string,
   717→): string | null {
   718→  if (!linkUrl) return null;
   719→  const baseUrl = process.env.NEXT_PUBLIC_APP_URL || fallbackBaseUrl;
   720→  // Remove trailing slash from base + leading slash from linkUrl to avoid double slashes
   721→  const cleanBase = baseUrl.replace(/\/+$/, "");
   722→  const cleanLink = linkUrl.startsWith("/") ? linkUrl : `/${linkUrl}`;
   723→  return `${cleanBase}${cleanLink}`;
   724→}
   725→
   726→/**
   727→ * Infer the audience (staff vs portal) from a linkUrl.
   728→ * Used by the email template to choose the right button label
   729→ * ("View in Dashboard" vs "View in Portal").
   730→ *
   731→ * @returns "portal" if the link starts with /portal, "staff" otherwise.
   732→ */
   733→export function inferLinkAudience(linkUrl: string | null | undefined): "staff" | "portal" {
   734→  if (!linkUrl) return "staff";
   735→  return linkUrl.startsWith("/portal") ? "portal" : "staff";
   736→}
