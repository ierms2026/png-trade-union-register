   310→function CreateUnionDialog({
   311→  onClose,
   312→  onCreated,
   313→}: {
   314→  onClose: () => void;
   315→  onCreated: () => void;
   316→}) {
   317→  const [unionName, setUnionName] = useState("");
   318→  const [registrationNo, setRegistrationNo] = useState("");
   319→  const [fileNumber, setFileNumber] = useState("");
   320→  const [status, setStatus] = useState("ACTIVE");
   321→  const [address, setAddress] = useState("");
   322→  const [contactPerson, setContactPerson] = useState("");
   323→  const [contactEmail, setContactEmail] = useState("");
   324→  const [contactPhone, setContactPhone] = useState("");
   325→  const [remarks, setRemarks] = useState("");
   326→  const [saving, setSaving] = useState(false);
   327→  const [touched, setTouched] = useState({ unionName: false, registrationNo: false });
   328→
   329→  const unionNameError = touched.unionName && !unionName.trim() ? "Union name is required." : null;
   330→  const regNoError = touched.registrationNo && !registrationNo.trim() ? "Registration number is required." : null;
   331→  const formValid = !!unionName.trim() && !!registrationNo.trim();
   332→
   333→  async function handleCreate(e: FormEvent) {
   334→    e.preventDefault();
   335→    setTouched({ unionName: true, registrationNo: true });
   336→    if (!formValid) return;
   337→
   338→    setSaving(true);
   339→    try {
   340→      const res = await fetch("/api/unions", {
   341→        method: "POST",
   342→        headers: { "Content-Type": "application/json" },
   343→        credentials: "include",
   344→        body: JSON.stringify({
   345→          unionName: unionName.trim(),
   346→          registrationNo: registrationNo.trim(),
   347→          fileNumber: fileNumber.trim() || null,
   348→          status,
   349→          address: address.trim() || null,
   350→          contactPerson: contactPerson.trim() || null,
   351→          contactEmail: contactEmail.trim() || null,
   352→          contactPhone: contactPhone.trim() || null,
   353→          remarks: remarks.trim() || null,
   354→        }),
   355→      });
   356→      const data = await res.json();
   357→      if (!res.ok) {
   358→        toast.error(data.error || "Failed to create union.");
   359→        return;
   360→      }
   361→      toast.success(`Union "${data.union.unionName}" registered successfully.`);
   362→      onCreated();
   363→    } catch {
   364→      toast.error("Network error. Please try again.");
   365→    } finally {
   366→      setSaving(false);
   367→    }
   368→  }
   369→
   370→  return (
   371→    <Dialog open onOpenChange={onClose}>
   372→      <DialogContent className="max-w-lg">
   373→        <DialogHeader>
   374→          <DialogTitle>Register New Union</DialogTitle>
   375→          <DialogDescription>
   376→            Create a new trade union record. The union will immediately appear in the public registry.
   377→          </DialogDescription>
   378→        </DialogHeader>
   379→        <form onSubmit={handleCreate} className="space-y-4">
   380→          {/* Required fields */}
   381→          <div className="space-y-2">
   382→            <Label htmlFor="unionName">Union Name *</Label>
   383→            <Input
   384→              id="unionName"
   385→              value={unionName}
   386→              onChange={(e) => setUnionName(e.target.value)}
   387→              onBlur={() => setTouched((p) => ({ ...p, unionName: true }))}
   388→              className={unionNameError ? "border-red-400" : ""}
   389→              placeholder="e.g. Papua Maritime Workers Union"
   390→              autoFocus
   391→            />
   392→            {unionNameError && <p className="text-xs text-red-600">{unionNameError}</p>}
   393→          </div>
   394→
   395→          <div className="grid grid-cols-2 gap-4">
   396→            <div className="space-y-2">
   397→              <Label htmlFor="registrationNo">Registration No *</Label>
   398→              <Input
   399→                id="registrationNo"
   400→                value={registrationNo}
   401→                onChange={(e) => setRegistrationNo(e.target.value)}
   402→                onBlur={() => setTouched((p) => ({ ...p, registrationNo: true }))}
   403→                className={regNoError ? "border-red-400" : ""}
   404→                placeholder="e.g. PNG-TU-0007"
   405→              />
   406→              {regNoError && <p className="text-xs text-red-600">{regNoError}</p>}
   407→            </div>
   408→            <div className="space-y-2">
   409→              <Label htmlFor="fileNumber">File Number</Label>
   410→              <Input
   411→                id="fileNumber"
   412→                value={fileNumber}
   413→                onChange={(e) => setFileNumber(e.target.value)}
   414→                placeholder="Optional"
   415→              />
   416→            </div>
   417→          </div>
   418→
   419→          <div className="space-y-2">
   420→            <Label>Status</Label>
   421→            <Select value={status} onValueChange={setStatus}>
   422→              <SelectTrigger>
   423→                <SelectValue />
   424→              </SelectTrigger>
   425→              <SelectContent>
   426→                <SelectItem value="ACTIVE">Active</SelectItem>
   427→                <SelectItem value="SUSPENDED">Suspended</SelectItem>
   428→                <SelectItem value="DEREGISTERED">Deregistered</SelectItem>
   429→              </SelectContent>
   430→            </Select>
   431→          </div>
   432→
   433→          {/* Contact info */}
   434→          <div className="grid grid-cols-2 gap-4">
   435→            <div className="space-y-2">
   436→              <Label htmlFor="contactPerson">Contact Person</Label>
   437→              <Input
   438→                id="contactPerson"
   439→                value={contactPerson}
   440→                onChange={(e) => setContactPerson(e.target.value)}
   441→                placeholder="Optional"
   442→              />
   443→            </div>
   444→            <div className="space-y-2">
   445→              <Label htmlFor="contactPhone">Contact Phone</Label>
   446→              <Input
   447→                id="contactPhone"
   448→                value={contactPhone}
   449→                onChange={(e) => setContactPhone(e.target.value)}
   450→                placeholder="Optional"
   451→              />
   452→            </div>
   453→          </div>
   454→
   455→          <div className="space-y-2">
   456→            <Label htmlFor="contactEmail">Contact Email</Label>
   457→            <Input
   458→              id="contactEmail"
   459→              type="email"
   460→              value={contactEmail}
   461→              onChange={(e) => setContactEmail(e.target.value)}
   462→              placeholder="Optional"
   463→            />
   464→          </div>
   465→
   466→          <div className="space-y-2">
   467→            <Label htmlFor="address">Address</Label>
   468→            <Textarea
   469→              id="address"
   470→              value={address}
   471→              onChange={(e) => setAddress(e.target.value)}
   472→              placeholder="Optional"
   473→              rows={2}
   474→            />
   475→          </div>
   476→
   477→          <div className="space-y-2">
   478→            <Label htmlFor="remarks">Remarks</Label>
   479→            <Textarea
   480→              id="remarks"
   481→              value={remarks}
   482→              onChange={(e) => setRemarks(e.target.value)}
   483→              placeholder="Internal notes (optional)"
   484→              rows={2}
   485→            />
   486→          </div>
   487→
   488→          <DialogFooter>
   489→            <Button type="button" variant="outline" onClick={onClose} disabled={saving}>
   490→              Cancel
   491→            </Button>
   492→            <Button type="submit" disabled={saving || !formValid}>
   493→              {saving ? (
   494→                <>
   495→                  <Loader2 className="mr-2 h-4 w-4 animate-spin" />
   496→                  Registering...
   497→                </>
   498→              ) : (
   499→                "Register Union"
   500→              )}
   501→            </Button>
   502→          </DialogFooter>
   503→        </form>
   504→      </DialogContent>
   505→    </Dialog>
   506→  );
   507→}

   510→function DocumentTile({
   511→  doc,
   512→  unionId,
   513→  onUploadVersion,
   514→}: {
   515→  doc: UnionDocument;
   516→  unionId: string;
   517→  onUploadVersion: () => void;
   518→}) {
   519→  return (
   520→    <div className="rounded-md border p-3">
   521→      <div className="flex items-start justify-between">
   522→        <div className="flex items-start gap-2">
   523→          <FileText className="mt-0.5 h-4 w-4 text-muted-foreground" />
   524→          <div>
   525→            <p className="text-sm font-medium">{doc.documentType}</p>
   526→            {doc.latestVersion ? (
   527→              <p className="mt-1 text-xs text-muted-foreground">
   528→                v{doc.latestVersion.versionNumber} · {new Date(doc.latestVersion.uploadedAt).toLocaleDateString()}
   529→              </p>
   530→            ) : (
   531→              <p className="mt-1 text-xs text-muted-foreground">No version uploaded</p>
   532→            )}
   533→          </div>
   534→        </div>
   535→      </div>
   536→      <div className="mt-3 flex gap-1">
   537→        <Button size="sm" variant="outline" className="h-7 text-xs" onClick={onUploadVersion}>
   538→          <Upload className="mr-1 h-3 w-3" />
   539→          New version
   540→        </Button>
   541→        {doc.latestVersion && (
   542→          <a href={`/api/unions/${unionId}/documents/${doc.id}/download`} target="_blank" rel="noopener noreferrer">
   543→            <Button size="sm" variant="ghost" className="h-7 text-xs">
   544→              <Download className="mr-1 h-3 w-3" />
   545→              Download
   546→            </Button>
   547→          </a>
   548→        )}
   549→      </div>
   550→    </div>
   551→  );
   552→}

   558→function EditUnionDialog({
   559→  union,
   560→  userRole,
   561→  onClose,
   562→  onSaved,
   563→}: {
   564→  union: Union;
   565→  userRole: string;
   566→  onClose: () => void;
   567→  onSaved: () => void;
   568→}) {
   569→  const isRegistrar = userRole === "REGISTRAR";
   570→  const isOfficer = userRole === "OFFICER";
   571→
   572→  const [unionName, setUnionName] = useState(union.unionName);
   573→  const [registrationNo, setRegistrationNo] = useState(union.registrationNo);
   574→  const [fileNumber, setFileNumber] = useState(union.fileNumber || "");
   575→  const [status, setStatus] = useState(union.status);
   576→  const [address, setAddress] = useState(union.address || "");
   577→  const [contactPerson, setContactPerson] = useState(union.contactPerson || "");
   578→  const [contactEmail, setContactEmail] = useState(union.contactEmail || "");
   579→  const [contactPhone, setContactPhone] = useState(union.contactPhone || "");
   580→  const [remarks, setRemarks] = useState(union.remarks || "");
   581→  const [saving, setSaving] = useState(false);
   582→
   583→  // Officer restricted fields (silently pinned server-side)
   584→  const restrictedFields = isOfficer
   585→    ? ["unionName", "registrationNo", "fileNumber", "status", "remarks"]
   586→    : [];
   587→
   588→  async function handleSave(e: FormEvent) {
   589→    e.preventDefault();
   590→    setSaving(true);
   591→    try {
   592→      const body: Record<string, unknown> = {};
   593→      if (isRegistrar) {
   594→        body.unionName = unionName;
   595→        body.registrationNo = registrationNo;
   596→        body.fileNumber = fileNumber || null;
   597→        body.status = status;
   598→        body.remarks = remarks || null;
   599→      }
   600→      body.address = address || null;
   601→      body.contactPerson = contactPerson || null;
   602→      body.contactEmail = contactEmail || null;
   603→      body.contactPhone = contactPhone || null;
   604→
   605→      const res = await fetch(`/api/unions/${union.id}`, {
   606→        method: "PUT",
   607→        headers: { "Content-Type": "application/json" },
   608→        credentials: "include",
   609→        body: JSON.stringify(body),
   610→      });
   611→      const data = await res.json();
   612→      if (!res.ok) {

   763→interface UploadDialogProps {
   764→  unionId: string;
   765→  defaultDocType: string;
   766→  open: boolean;
   767→  onClose: () => void;
   768→  onUploaded: () => void;
   769→}
   770→
   771→function UploadDialog({ unionId, defaultDocType, open, onClose, onUploaded }: UploadDialogProps) {
   772→  const [documentType, setDocumentType] = useState(defaultDocType);
   773→  const [title, setTitle] = useState("");
   774→  const [file, setFile] = useState<File | null>(null);
   775→  const [uploading, setUploading] = useState(false);
   776→
   777→  // Sync the doc type when the default changes (e.g., user clicks "New version" on a specific doc)
   778→  useState(() => {
   779→    if (defaultDocType) setDocumentType(defaultDocType);
   780→  });
   781→
   782→  async function handleUpload(e: FormEvent) {
   783→    e.preventDefault();
   784→    if (!file) {
   785→      toast.error("Please select a file.");
   786→      return;
   787→    }
   788→    if (!documentType.trim()) {
   789→      toast.error("Document type is required.");
   790→      return;
   791→    }
   792→    if (!title.trim()) {
   793→      toast.error("Title is required.");
   794→      return;
   795→    }
   796→    setUploading(true);
   797→    try {
   798→      const formData = new FormData();
   799→      formData.append("file", file);
   800→      formData.append("documentType", documentType.trim());
   801→      formData.append("title", title.trim());
   802→      const res = await fetch(`/api/unions/${unionId}/documents`, {
   803→        method: "POST",
   804→        credentials: "include",
   805→        body: formData,
   806→      });
   807→      const data = await res.json();
   808→      if (!res.ok) {
   809→        toast.error(data.error || "Upload failed.");
   810→        return;
   811→      }
   812→      toast.success(`Uploaded ${documentType} v${data.version.versionNumber}.`);
   813→      onUploaded();
   814→    } catch {
   815→      toast.error("Network error. Please try again.");
   816→    } finally {
   817→      setUploading(false);
   818→    }
   819→  }
   820→
   821→  return (
   822→    <Dialog open={open} onOpenChange={(o) => !o && onClose()}>
   823→      <DialogContent>
   824→        <DialogHeader>
   825→          <DialogTitle>Upload Document</DialogTitle>
   826→          <DialogDescription>
   827→            Upload a new version of a document. Duplicate detection is based on SHA-256 checksum.
   828→          </DialogDescription>
   829→        </DialogHeader>
   830→        <form onSubmit={handleUpload} className="space-y-4">
   831→          <div className="space-y-2">
   832→            <Label htmlFor="docType">Document Type</Label>
   833→            <Input
   834→              id="docType"
   835→              value={documentType}
   836→              onChange={(e) => setDocumentType(e.target.value)}
   837→              placeholder="e.g. Constitution, Financial Return"
   838→            />
   839→          </div>
   840→          <div className="space-y-2">
   841→            <Label htmlFor="title">Title</Label>
   842→            <Input
   843→              id="title"
   844→              value={title}
   845→              onChange={(e) => setTitle(e.target.value)}
   846→              placeholder="A descriptive title for this version"
   847→            />
   848→          </div>
   849→          <div className="space-y-2">
   850→            <Label htmlFor="file">File</Label>
   851→            <Input
   852→              id="file"
   853→              type="file"
   854→              onChange={(e) => setFile(e.target.files?.[0] ?? null)}
   855→              accept=".pdf,.doc,.docx,.jpg,.png"
   856→            />
   857→            {file && (
   858→              <p className="text-xs text-muted-foreground">
   859→                {file.name} ({(file.size / 1024).toFixed(1)} KB)
   860→              </p>
   861→            )}
   862→          </div>
   863→          <DialogFooter>
   864→            <Button type="button" variant="outline" onClick={onClose} disabled={uploading}>
   865→              Cancel
   866→            </Button>
   867→            <Button type="submit" disabled={uploading || !file}>
   868→              {uploading ? (
   869→                <>
   870→                  <Loader2 className="mr-2 h-4 w-4 animate-spin" />
   871→                  Uploading...
   872→                </>
   873→              ) : (
   874→                <>
   875→                  <Upload className="mr-2 h-4 w-4" />
   876→                  Upload
   877→                </>
   878→              )}
   879→            </Button>
   880→          </DialogFooter>
   881→        </form>
   882→      </DialogContent>
   883→    </Dialog>
   884→  );
   885→}
