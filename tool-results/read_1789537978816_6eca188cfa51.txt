    93→export function UnionsView({ user, initialUnionId }: { user: { role: string; assignedUnionId: string | null }; initialUnionId?: string | null }) {

    96→  const [uploadDialogOpen, setUploadDialogOpen] = useState(false);
    97→  const [uploadDocType, setUploadDocType] = useState<string>("");
    98→  const [editDialogOpen, setEditDialogOpen] = useState(false);
    99→  const [searchQuery, setSearchQuery] = useState("");
   100→
   101→  const selectedUnion = data?.unions.find((u) => u.id === selectedUnionId);
   102→
   103→  // Client-side filter — same pattern as the public portal. Instant, no
   104→  // network round-trip. Filters by union name or registration number.
   105→  const filteredUnions = (() => {
   106→    const all = data?.unions ?? [];
   107→    const q = searchQuery.trim().toLowerCase();
   108→    if (!q) return all;
   109→    return all.filter(
   110→      (u) =>
   111→        u.unionName.toLowerCase().includes(q) ||
   112→        u.registrationNo.toLowerCase().includes(q),
   113→    );
   114→  })();

   116→  if (loading) return <CardListSkeleton count={6} />;
   117→  if (error) return <ErrorState description={error} onRetry={refetch} />;

   119→    return (
   120→      <EmptyState
   121→        icon={Building2}
   122→        title="No unions registered"
   123→        description="Once unions are registered, they will appear here."
   124→      />
   125→    );
   126→  }
   127→
   128→  // --- Union detail view ---
   129→  if (selectedUnion) {
   130→    return (
   131→      <div className="space-y-6">
   132→        <Button variant="ghost" size="sm" onClick={() => setSelectedUnionId(null)}>
   133→          <ArrowLeft className="mr-2 h-4 w-4" />
   134→          Back to unions
   135→        </Button>
   136→
   137→        <Card>
   138→          <CardHeader>
   139→            <div className="flex items-start justify-between">

   141→                <CardTitle className="text-xl">{selectedUnion.unionName}</CardTitle>
   142→                <CardDescription className="mt-1 flex items-center gap-2">

   144→                  <Badge variant={selectedUnion.status === "ACTIVE" ? "default" : "secondary"}>

   146→                  </Badge>
   147→                </CardDescription>
   148→              </div>

   151→                  size="sm"
   152→                  variant="outline"
   153→                  onClick={() => setEditDialogOpen(true)}
   154→                >
   155→                  <Pencil className="mr-2 h-4 w-4" />

   157→                </Button>
   158→                <Button
   159→                  size="sm"
   160→                  onClick={() => {

   162→                  }}
   163→                >
   164→                  <Upload className="mr-2 h-4 w-4" />
   165→                  Upload Document
   166→                </Button>
   167→              </div>
   168→            </div>
   169→          </CardHeader>
   170→          <CardContent className="space-y-4">
   171→            {/* Contact info */}
   172→            <div className="grid gap-4 text-sm md:grid-cols-2">
   173→              {selectedUnion.fileNumber && (
   174→                <div>
   175→                  <p className="text-xs text-muted-foreground">File Number</p>
   176→                  <p className="font-medium">{selectedUnion.fileNumber}</p>
   177→                </div>
   178→              )}

   180→                <div>

   184→              )}

   325→function DocumentTile({
   326→  doc,
   327→  unionId,
   328→  userRole,
   329→  onUploadVersion,
   330→}: {
   331→  doc: UnionDocument;
   332→  unionId: string;
   333→  userRole: string;
   334→  onUploadVersion: () => void;
   335→}) {
   336→  const [versionsOpen, setVersionsOpen] = useState(false);
   337→  const [versions, setVersions] = useState<unknown[] | null>(null);
   338→  const [versionsLoading, setVersionsLoading] = useState(false);
   339→  const [previewDoc, setPreviewDoc] = useState<{
   340→    documentType: string;
   341→    version: number;
   342→    fileName: string;
   343→    fileExtension: string | null;
   344→    downloadUrl: string;
   345→  } | null>(null);
   346→
   347→  const canMutate = userRole !== "INSPECTOR";
   348→
   349→  async function loadVersions() {
   350→    setVersionsLoading(true);
   351→    try {
   352→      const res = await fetch(`/api/unions/${unionId}/documents/${doc.id}/versions`, {
   353→        credentials: "include",
   354→      });
   355→      const data = await res.json();
   356→      if (res.ok) {
   357→        setVersions(data.versions);
   358→      } else {
   359→        toast.error("Failed to load versions.");
   360→      }
   361→    } catch {
   362→      toast.error("Network error.");
   363→    } finally {
   364→      setVersionsLoading(false);
   365→    }
   366→  }
   367→
   368→  async function restoreVersion(versionId: string) {
   369→    try {
   370→      const res = await fetch(`/api/unions/${unionId}/documents/${doc.id}/restore`, {
   371→        method: "POST",
   372→        headers: { "Content-Type": "application/json" },
   373→        credentials: "include",
   374→        body: JSON.stringify({ versionId }),
   375→      });
   376→      const data = await res.json();
   377→      if (!res.ok) {
   378→        toast.error(data.error || "Restore failed.");
   379→        return;
   380→      }
   381→      toast.success(`Restored from version ${data.restoredFromVersion}. New version: ${data.newVersion.versionNumber}.`);
   382→      setVersionsOpen(false);
   383→      // Reload the union list to reflect the new version
   384→      window.location.reload();
   385→    } catch {
   386→      toast.error("Network error.");
   387→    }
   388→  }
   389→
   390→  return (
   391→    <div className="rounded-md border p-3">
   392→      <div className="flex items-start justify-between">
   393→        <div className="flex items-start gap-2">
   394→          <FileText className="mt-0.5 h-4 w-4 text-muted-foreground" />
   395→          <div>
   396→            <p className="text-sm font-medium">{doc.documentType}</p>
   397→            {doc.disposedAt ? (
   398→              <Badge variant="destructive" className="mt-1 text-xs">Disposed</Badge>
   399→            ) : doc.latestVersion ? (
   400→              <div className="mt-1 flex flex-wrap items-center gap-2">
   401→                <p className="text-xs text-muted-foreground">
   402→                  v{doc.latestVersion.versionNumber} · {new Date(doc.latestVersion.uploadedAt).toLocaleDateString()}
   403→                </p>
   404→                {doc.reviewStatus === "PENDING" && (
   405→                  <Badge variant="secondary" className="text-xs bg-amber-100 text-amber-800 border-amber-200">Pending Review</Badge>
   406→                )}
   407→                {doc.reviewStatus === "APPROVED" && (
   408→                  <Badge variant="secondary" className="text-xs bg-green-100 text-green-800 border-green-200">Approved</Badge>
   409→                )}
   410→                {doc.reviewStatus === "REJECTED" && (
   411→                  <Badge variant="secondary" className="text-xs bg-red-100 text-red-800 border-red-200 cursor-help" title={doc.rejectionReason ?? "Rejected by Registrar"}>
   412→                    Rejected
   413→                  </Badge>
   414→                )}
   415→              </div>
   416→            ) : (
   417→              <p className="mt-1 text-xs text-muted-foreground">No version uploaded</p>
   418→            )}
   419→            {doc.reviewStatus === "REJECTED" && doc.rejectionReason && (
   420→              <p className="mt-1 text-xs text-red-600">Reason: {doc.rejectionReason}</p>
   421→            )}
   422→          </div>
   423→        </div>
   424→      </div>
   425→      <div className="mt-3 flex flex-wrap gap-1">
   426→        {canMutate && (
   427→          <Button size="sm" variant="outline" className="h-7 text-xs" onClick={onUploadVersion}>
   428→            <Upload className="mr-1 h-3 w-3" />
   429→            New version
   430→          </Button>
   431→        )}
   432→        {doc.latestVersion && !doc.disposedAt && (
   433→          <>
   434→            <Button
   435→              size="sm"
   436→              variant="ghost"
   437→              className="h-7 text-xs"
   438→              onClick={() =>
   439→                setPreviewDoc({
   440→                  documentType: doc.documentType,
   441→                  version: doc.latestVersion!.versionNumber,
   442→                  fileName: doc.latestVersion!.fileName,
   443→                  fileExtension: doc.latestVersion!.fileExtension,
   444→                  downloadUrl: `/api/unions/${unionId}/documents/${doc.id}/download`,
   445→                })
   446→              }
   447→            >
   448→              <Eye className="mr-1 h-3 w-3" />
   449→              View
   450→            </Button>
   451→            <a href={`/api/unions/${unionId}/documents/${doc.id}/download`} target="_blank" rel="noopener noreferrer">
   452→              <Button size="sm" variant="ghost" className="h-7 text-xs">
   453→                <Download className="mr-1 h-3 w-3" />
   454→                Download
   455→              </Button>
   456→            </a>
   457→          </>
   458→        )}
   459→        <Button
   460→          size="sm"
   461→          variant="ghost"
   462→          className="h-7 text-xs"
   463→          onClick={() => {
   464→            if (!versionsOpen && !versions) loadVersions();
   465→            setVersionsOpen(!versionsOpen);
   466→          }}
   467→        >
   468→          <History className="mr-1 h-3 w-3" />
   469→          History
   470→        </Button>
   471→      </div>
   472→
   473→      {versionsOpen && (
   474→        <div className="mt-3 border-t pt-2">
   475→          {versionsLoading ? (
   476→            <p className="text-xs text-muted-foreground">Loading...</p>
   477→          ) : versions && versions.length > 0 ? (
   478→            <div className="space-y-1">
   479→              {(versions as Array<{
   480→                id: string;
   481→                versionNumber: number;
   482→                fileName: string;
   483→                uploadedAt: string;
   484→                isCurrent: boolean;
   485→              }>).map((v) => (
   486→                <div
   487→                  key={v.id}
   488→                  className="flex items-center justify-between rounded bg-muted/30 p-2 text-xs"
   489→                >
   490→                  <div>
   491→                    <span className="font-medium">v{v.versionNumber}</span>
   492→                    {v.isCurrent && <Badge className="ml-1 text-[10px]">Current</Badge>}
   493→                    <p className="text-muted-foreground">{new Date(v.uploadedAt).toLocaleString()}</p>
   494→                  </div>
   495→                  {!v.isCurrent && canMutate && (
   496→                    <Button
   497→                      size="sm"
   498→                      variant="ghost"
   499→                      className="h-6 text-xs"
   500→                      onClick={() => restoreVersion(v.id)}
   501→                    >
   502→                      <RotateCcw className="mr-1 h-3 w-3" />
   503→                      Restore
   504→                    </Button>
   505→                  )}
   506→                </div>
   507→              ))}
   508→            </div>
   509→          ) : (
   510→            <p className="text-xs text-muted-foreground">No versions.</p>
   511→          )}
   512→        </div>
   513→      )}
   514→
   515→      {previewDoc && (
   516→        <DocumentPreviewModal
   517→          open={!!previewDoc}
   518→          onOpenChange={(open) => !open && setPreviewDoc(null)}
   519→          documentType={previewDoc.documentType}
   520→          version={previewDoc.version}
   521→          fileName={previewDoc.fileName}
   522→          fileExtension={previewDoc.fileExtension}
   523→          downloadUrl={previewDoc.downloadUrl}
   524→        />
   525→      )}
   526→    </div>
   527→  );
   528→}

   534→function EditUnionDialog({
   535→  union,
   536→  userRole,
   537→  onClose,
   538→  onSaved,
   539→}: {
   540→  union: Union;
   541→  userRole: string;
   542→  onClose: () => void;
   543→  onSaved: () => void;
   544→}) {
   545→  const isRegistrar = userRole === "REGISTRAR";
   546→  const isOfficer = userRole === "OFFICER";
   547→
   548→  const [unionName, setUnionName] = useState(union.unionName);
   549→  const [registrationNo, setRegistrationNo] = useState(union.registrationNo);
   550→  const [fileNumber, setFileNumber] = useState(union.fileNumber || "");
   551→  const [status, setStatus] = useState(union.status);
   552→  const [address, setAddress] = useState(union.address || "");
   553→  const [contactPerson, setContactPerson] = useState(union.contactPerson || "");
   554→  const [contactEmail, setContactEmail] = useState(union.contactEmail || "");
   555→  const [contactPhone, setContactPhone] = useState(union.contactPhone || "");
   556→  const [remarks, setRemarks] = useState(union.remarks || "");
   557→  const [saving, setSaving] = useState(false);
   558→
   559→  // Officer restricted fields (silently pinned server-side)
   560→  const restrictedFields = isOfficer
   561→    ? ["unionName", "registrationNo", "fileNumber", "status", "remarks"]
   562→    : [];
   563→
   564→  async function handleSave(e: FormEvent) {
   565→    e.preventDefault();
   566→    setSaving(true);
   567→    try {
   568→      const body: Record<string, unknown> = {};
   569→      if (isRegistrar) {
   570→        body.unionName = unionName;
   571→        body.registrationNo = registrationNo;
   572→        body.fileNumber = fileNumber || null;
   573→        body.status = status;
   574→        body.remarks = remarks || null;
   575→      }
   576→      body.address = address || null;
   577→      body.contactPerson = contactPerson || null;
   578→      body.contactEmail = contactEmail || null;
   579→      body.contactPhone = contactPhone || null;
   580→
   581→      const res = await fetch(`/api/unions/${union.id}`, {

   739→interface UploadDialogProps {
   740→  unionId: string;
   741→  defaultDocType: string;
   742→  open: boolean;
   743→  onClose: () => void;
   744→  onUploaded: () => void;
   745→}
   746→
   747→function UploadDialog({ unionId, defaultDocType, open, onClose, onUploaded }: UploadDialogProps) {
   748→  const [documentType, setDocumentType] = useState(defaultDocType);
   749→  const [title, setTitle] = useState("");
   750→  const [file, setFile] = useState<File | null>(null);
   751→  const [uploading, setUploading] = useState(false);
   752→
   753→  // Sync the doc type when the default changes (e.g., user clicks "New version" on a specific doc)
   754→  useState(() => {
   755→    if (defaultDocType) setDocumentType(defaultDocType);
   756→  });
   757→
   758→  async function handleUpload(e: FormEvent) {
   759→    e.preventDefault();
   760→    if (!file) {
   761→      toast.error("Please select a file.");
   762→      return;
   763→    }
   764→    if (!documentType.trim()) {
   765→      toast.error("Document type is required.");
   766→      return;
   767→    }
   768→    if (!title.trim()) {
   769→      toast.error("Title is required.");
   770→      return;
   771→    }
   772→    setUploading(true);
   773→    try {
   774→      const formData = new FormData();
   775→      formData.append("file", file);
   776→      formData.append("documentType", documentType.trim());
   777→      formData.append("title", title.trim());
   778→      const res = await fetch(`/api/unions/${unionId}/documents`, {
   779→        method: "POST",
   780→        credentials: "include",
   781→        body: formData,
   782→      });
   783→      const data = await res.json();
   784→      if (!res.ok) {
   785→        toast.error(data.error || "Upload failed.");
   786→        return;
   787→      }
   788→      toast.success(`Uploaded ${documentType} v${data.version.versionNumber}.`);
   789→      onUploaded();
   790→    } catch {
   791→      toast.error("Network error. Please try again.");
   792→    } finally {
   793→      setUploading(false);
   794→    }
   795→  }
   796→
   797→  return (
   798→    <Dialog open={open} onOpenChange={(o) => !o && onClose()}>
   799→      <DialogContent>
   800→        <DialogHeader>
   801→          <DialogTitle>Upload Document</DialogTitle>
   802→          <DialogDescription>
   803→            Upload a new version of a document. Duplicate detection is based on SHA-256 checksum.
   804→          </DialogDescription>
   805→        </DialogHeader>
   806→        <form onSubmit={handleUpload} className="space-y-4">
   807→          <div className="space-y-2">
   808→            <Label htmlFor="docType">Document Type</Label>
   809→            <Input
   810→              id="docType"
   811→              value={documentType}
   812→              onChange={(e) => setDocumentType(e.target.value)}
   813→              placeholder="e.g. Constitution, Financial Return"
   814→            />
   815→          </div>
   816→          <div className="space-y-2">
   817→            <Label htmlFor="title">Title</Label>
   818→            <Input
   819→              id="title"
   820→              value={title}
   821→              onChange={(e) => setTitle(e.target.value)}
   822→              placeholder="A descriptive title for this version"
   823→            />
   824→          </div>
   825→          <div className="space-y-2">
   826→            <Label htmlFor="file">File</Label>
   827→            <Input
   828→              id="file"
   829→              type="file"
   830→              onChange={(e) => setFile(e.target.files?.[0] ?? null)}
   831→              accept=".pdf,.doc,.docx,.jpg,.png"
   832→            />
   833→            {file && (
   834→              <p className="text-xs text-muted-foreground">
   835→                {file.name} ({(file.size / 1024).toFixed(1)} KB)
   836→              </p>
   837→            )}
   838→          </div>
   839→          <DialogFooter>
   840→            <Button type="button" variant="outline" onClick={onClose} disabled={uploading}>
   841→              Cancel
   842→            </Button>
   843→            <Button type="submit" disabled={uploading || !file}>
   844→              {uploading ? (
   845→                <>
   846→                  <Loader2 className="mr-2 h-4 w-4 animate-spin" />
   847→                  Uploading...
   848→                </>
   849→              ) : (
   850→                <>
   851→                  <Upload className="mr-2 h-4 w-4" />
   852→                  Upload
   853→                </>
   854→              )}
   855→            </Button>
   856→          </DialogFooter>
   857→        </form>
   858→      </DialogContent>
   859→    </Dialog>
   860→  );
   861→}
