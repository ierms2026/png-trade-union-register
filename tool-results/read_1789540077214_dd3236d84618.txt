   345→   345→function DocumentTile({
   346→   346→  doc,
   347→   347→  unionId,
   348→   348→  userRole,
   349→   349→  onUploadVersion,
   350→   350→}: {
   351→   351→  doc: UnionDocument;
   352→   352→  unionId: string;
   353→   353→  userRole: string;
   354→   354→  onUploadVersion: () => void;
   355→   355→}) {
   356→   356→  const [versionsOpen, setVersionsOpen] = useState(false);
   357→   357→  const [versions, setVersions] = useState<unknown[] | null>(null);
   358→   358→  const [versionsLoading, setVersionsLoading] = useState(false);
   359→   359→  const [previewDoc, setPreviewDoc] = useState<{
   360→   360→    documentType: string;
   361→   361→    version: number;
   362→   362→    fileName: string;
   363→   363→    fileExtension: string | null;
   364→   364→    downloadUrl: string;
   365→   365→  } | null>(null);
   366→   366→
   367→   367→  const canMutate = userRole !== "INSPECTOR";
   368→   368→
   369→   369→  async function loadVersions() {
   370→   370→    setVersionsLoading(true);
   371→   371→    try {
   372→   372→      const res = await fetch(`/api/unions/${unionId}/documents/${doc.id}/versions`, {
   373→   373→        credentials: "include",
   374→   374→      });
   375→   375→      const data = await res.json();
   376→   376→      if (res.ok) {
   377→   377→        setVersions(data.versions);
   378→   378→      } else {
   379→   379→        toast.error("Failed to load versions.");
   380→   380→      }
   381→   381→    } catch {
   382→   382→      toast.error("Network error.");
   383→   383→    } finally {
   384→   384→      setVersionsLoading(false);
   385→   385→    }
   386→   386→  }
   387→   387→
   388→   388→  async function restoreVersion(versionId: string) {
   389→   389→    try {
   390→   390→      const res = await fetch(`/api/unions/${unionId}/documents/${doc.id}/restore`, {
   391→   391→        method: "POST",
   392→   392→        headers: { "Content-Type": "application/json" },
   393→   393→        credentials: "include",
   394→   394→        body: JSON.stringify({ versionId }),
   395→   395→      });
   396→   396→      const data = await res.json();
   397→   397→      if (!res.ok) {
   398→   398→        toast.error(data.error || "Restore failed.");
   399→   399→        return;
   400→   400→      }
   401→   401→      toast.success(`Restored from version ${data.restoredFromVersion}. New version: ${data.newVersion.versionNumber}.`);
   402→   402→      setVersionsOpen(false);
   403→   403→      // Reload the union list to reflect the new version
   404→   404→      window.location.reload();
   405→   405→    } catch {
   406→   406→      toast.error("Network error.");
   407→   407→    }
   408→   408→  }
   409→   409→
   410→   410→  return (
   411→   411→    <div className="rounded-md border p-3">
   412→   412→      <div className="flex items-start justify-between">
   413→   413→        <div className="flex items-start gap-2">
   414→   414→          <FileText className="mt-0.5 h-4 w-4 text-muted-foreground" />
   415→   415→          <div>
   416→   416→            <p className="text-sm font-medium">{doc.documentType}</p>
   417→   417→            {doc.disposedAt ? (
   418→   418→              <Badge variant="destructive" className="mt-1 text-xs">Disposed</Badge>
   419→   419→            ) : doc.latestVersion ? (
   420→   420→              <div className="mt-1 flex flex-wrap items-center gap-2">
   421→   421→                <p className="text-xs text-muted-foreground">
   422→   422→                  v{doc.latestVersion.versionNumber} · {new Date(doc.latestVersion.uploadedAt).toLocaleDateString()}
   423→   423→                </p>
   424→   424→                {doc.reviewStatus === "PENDING" && (
   425→   425→                  <Badge variant="secondary" className="text-xs bg-amber-100 text-amber-800 border-amber-200">Pending Review</Badge>
   426→   426→                )}
   427→   427→                {doc.reviewStatus === "APPROVED" && (
   428→   428→                  <Badge variant="secondary" className="text-xs bg-green-100 text-green-800 border-green-200">Approved</Badge>
   429→   429→                )}
   430→   430→                {doc.reviewStatus === "REJECTED" && (
   431→   431→                  <Badge variant="secondary" className="text-xs bg-red-100 text-red-800 border-red-200 cursor-help" title={doc.rejectionReason ?? "Rejected by Registrar"}>
   432→   432→                    Rejected
   433→   433→                  </Badge>
   434→   434→                )}
   435→   435→              </div>
   436→   436→            ) : (
   437→   437→              <p className="mt-1 text-xs text-muted-foreground">No version uploaded</p>
   438→   438→            )}
   439→   439→            {doc.reviewStatus === "REJECTED" && doc.rejectionReason && (
   440→   440→              <p className="mt-1 text-xs text-red-600">Reason: {doc.rejectionReason}</p>
   441→   441→            )}
   442→   442→          </div>
   443→   443→        </div>
   444→   444→      </div>
   445→   445→      <div className="mt-3 flex flex-wrap gap-1">
   446→   446→        {canMutate && (
   447→   447→          <Button size="sm" variant="outline" className="h-7 text-xs" onClick={onUploadVersion}>
   448→   448→            <Upload className="mr-1 h-3 w-3" />
   449→   449→            New version
   450→   450→          </Button>
   451→   451→        )}
   452→   452→        {doc.latestVersion && !doc.disposedAt && (
   453→   453→          <>
   454→   454→            <Button
   455→   455→              size="sm"
   456→   456→              variant="ghost"
   457→   457→              className="h-7 text-xs"
   458→   458→              onClick={() =>
   459→   459→                setPreviewDoc({
   460→   460→                  documentType: doc.documentType,
   461→   461→                  version: doc.latestVersion!.versionNumber,
   462→   462→                  fileName: doc.latestVersion!.fileName,
   463→   463→                  fileExtension: doc.latestVersion!.fileExtension,
   464→   464→                  downloadUrl: `/api/unions/${unionId}/documents/${doc.id}/download`,
   465→   465→                })
   466→   466→              }
   467→   467→            >
   468→   468→              <Eye className="mr-1 h-3 w-3" />
   469→   469→              View
   470→   470→            </Button>
   471→   471→            <a href={`/api/unions/${unionId}/documents/${doc.id}/download`} target="_blank" rel="noopener noreferrer">
   472→   472→              <Button size="sm" variant="ghost" className="h-7 text-xs">
   473→   473→                <Download className="mr-1 h-3 w-3" />
   474→   474→                Download
