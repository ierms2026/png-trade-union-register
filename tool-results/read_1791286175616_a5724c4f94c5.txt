   300→// =============================================================================
   301→// Sparkline — chunk 4.2b. Decorative shape-only polyline for KPI cards.
   302→//
   303→// Auto-scaled; no axes, no numbers (the card's value IS the number). Hidden
   304→// by the caller when hasSufficientHistory() is false. Marked aria-hidden
   305→// (the adjacent card value is the accessible number — the sparkline is
   306→// decorative context). Full history remains accessible via the TrendChart's
   307→// visually-hidden data table.
   308→//
   309→// GEOMETRY: delegated to `buildSparklinePoints` in snapshot.ts (pure). This
   310→// component just renders the SVG shell with the polyline.
   311→//
   312→// GAP HANDLING: null values in the series are skipped by `buildSparklinePoints`
   313→// — the polyline jumps from the previous non-null point to the next, leaving
   314→// a visual gap (consistent with the main chart's connectNulls={false}).
   315→// =============================================================================
   316→
   317→interface SparklineProps {
   318→  /** Numeric series (with nulls for gaps). Built via `extractSparklineSeries`. */
   319→  values: (number | null)[];
   320→  /** Pixel width of the sparkline SVG. Defaults to 100. */
   321→  width?: number;
   322→  /** Pixel height of the sparkline SVG. Defaults to 24. */
   323→  height?: number;
   324→  /** Stroke color (CSS var token). Defaults to currentColor. */
   325→  stroke?: string;
   326→  /** Stroke width. Defaults to 1.5. */
   327→  strokeWidth?: number;
   328→}
   329→
   330→function Sparkline({
   331→  values,
   332→  width = 100,
   333→  height = 24,
   334→  stroke = "currentColor",
   335→  strokeWidth = 1.5,
   336→}: SparklineProps) {
   337→  const points = buildSparklinePoints(values, width, height);
   338→  if (!points) return null;
   339→
   340→  return (
   341→    <svg
   342→      width={width}
   343→      height={height}
   344→      viewBox={`0 0 ${width} ${height}`}
   345→      fill="none"
   346→      // aria-hidden is set by the parent StatCard wrapper. We don't repeat
   347→      // it here because the sparkline may also be used in non-StatCard
   348→      // contexts where the parent SHOULD set aria-hidden — the convention
   349→      // is "decorative sparkline = parent marks aria-hidden".
   350→      role="presentation"
   351→      focusable={false}
   352→    >
   353→      <polyline
   354→        points={points}
   355→        stroke={stroke}
   356→        strokeWidth={strokeWidth}
   357→        strokeLinejoin="round"
   358→        strokeLinecap="round"
   359→      />
   360→    </svg>
   361→  );
   362→}

   364→// =============================================================================
   365→// TrendChart — chunk 4.2b. Compliance-rate line chart card.
   366→//
   367→// TITLE: "Compliance Trend" (6.2 may rename — see handoff).
   368→// WINDOW: most recent TREND_WINDOW_DAYS (default 365) of available history,
   369→//   derived from point dates via `sliceTrendToWindow` — never hardcoded.
   370→// RENDERING RULES:
   371→//   - Line breaks at gaps (missing days). `buildChartSeriesWithGaps` inserts
   372→//     explicit null-percentage rows for missing intermediate dates; recharts
   373→//     with connectNulls={false} (default) renders the break.
   374→//   - Y-axis fixed 0–100% with gridlines at 0/25/50/75/100 (honesty > drama).
   375→//   - X-axis labels via the shared date util (formatAbsolute, abbreviated).
   376→//   - Hover tooltips per point: "12 Jun 2026 · 17 of 78 (22%)" — same phrasing
   377→//     as the 1.2 rate card.
   378→//   - Denominator changes across the window (required types or unions
   379→//     changed): the jump is HONEST — rendered as-is. Tooltips self-explain
   380→//     via numerator/denominator. No visual annotation added (see handoff
   381→//     for the decision + counter-proposal).
   382→// INSUFFICIENT HISTORY: hasSufficientHistory() false (0 or 1 points) →
   383→//   placeholder "Trend will appear as daily snapshots are collected."
   384→//   Never an error look, never a fake line. Exactly 2 points → render the
   385→//   chart (degenerate but honest).
   386→// ACCESSIBILITY:
   387→//   - SVG wrapped in a container with role="img" + concise aria-label
   388→//     ("Compliance rate trend, last N days").
   389→//   - Visually-hidden HTML table (date, counts, rate) carries the full
   390→//     series for screen readers — the guaranteed accessible path. Hover
   391→//     tooltips are a mouse bonus, not the accessible path.
   392→//   - Per-point keyboard focus NOT provided (see handoff for the decision
   393→//     + counter-proposal — recharts does not expose per-point focus
   394→//     natively; a custom wrapper would have to inject it).
   395→// =============================================================================
   396→
   397→interface TrendChartProps {
   398→  /** Full trend points (as returned by /api/dashboard/trend). The chart will
   399→   * slice to the window itself via sliceTrendToWindow. */
   400→  points: ComplianceTrendPoint[];
   401→  /** Loading state — when true, render a skeleton placeholder. */
   402→  loading?: boolean;
   403→  /** Error message — when present, render an error state with retry. */
   404→  error?: string | null;
   405→  /** Retry callback for the error state's retry button. */
   406→  onRetry?: () => void;
   407→}

   409→function TrendChart({ points, loading, error, onRetry }: TrendChartProps) {
   410→  // --- 1. Slice to the configured window (most recent N days of history) ---
   411→  // Per 4.2b spec: "Window: most recent TREND_WINDOW_DAYS of available
   412→  // history — derived from point dates, never hardcoded." The window is
   413→  // measured from the LAST point's date, NOT from `new Date()` — so a
   414→  // pipeline that's been down for a week still shows the last year of
   415→  // valid data, not "no recent data".
   416→  const windowedPoints = sliceTrendToWindow(points, TREND_WINDOW_DAYS);
   417→
   418→  // --- 2. Build the chart data array with explicit gap markers ---
   419→  // `buildChartSeriesWithGaps` inserts null-percentage rows for missing
   420→  // intermediate dates. With connectNulls={false} (default), recharts
   421→  // renders a LINE BREAK at those rows — no interpolation, no synthetic
   422→  // connecting segments.
   423→  const chartData: TrendChartDatum[] = buildChartSeriesWithGaps(windowedPoints);
   424→
   425→  // --- 3. Decide: placeholder or chart? ---
   426→  // hasSufficientHistory() === false (0 or 1 points) → placeholder.
   427→  // Exactly 2 points → render (degenerate but honest — one line segment).
   428→  const sufficient = hasSufficientHistory(windowedPoints, 2);
   429→  const windowDays = TREND_WINDOW_DAYS;
   430→
   431→  return (
   432→    <Card>
   433→      <CardHeader>
   434→        <div className="flex items-center justify-between">
   435→          <div>
   436→            <CardTitle className="text-lg">Compliance Trend</CardTitle>
   437→            <CardDescription>
   438→              Daily compliance rate over the last {windowDays} days of collected history.
   439→            </CardDescription>
   440→          </div>
   441→          {sufficient && (
   442→            <a href="/api/export/trend" target="_blank" rel="noopener noreferrer">
   443→              <Button variant="outline" size="sm" className="gap-1.5">
   444→                <FileDown className="h-3.5 w-3.5" />
   445→                Export CSV
   446→              </Button>
   447→            </a>
   448→          )}
   449→        </div>
   450→      </CardHeader>
   451→      <CardContent>
   452→        {loading ? (
   453→          <div className="h-[240px] animate-pulse rounded-md bg-muted" />
   454→        ) : error ? (
   455→          <ErrorState description={error} onRetry={onRetry} />
   456→        ) : !sufficient ? (
   457→          // --- PLACEHOLDER: 0 or 1 points ---
   458→          // First-class state, not an error. The pipeline is fresh; the
   459→          // chart will populate as daily snapshots are collected (the
   460→          // /api/cron/compliance-snapshot route fires daily at 7:55 UTC).
   461→          // Worded per 4.2b spec ("Trend will appear as daily snapshots
   462→          // are collected.") — 6.2 may adjust the wording.
   463→          <div className="flex h-[240px] flex-col items-center justify-center text-center">
   464→            <div className="rounded-md bg-muted/50 px-4 py-3">
   465→              <p className="text-sm font-medium text-muted-foreground">
   466→                Trend will appear as daily snapshots are collected.
   467→              </p>
   468→              <p className="mt-1 text-xs text-muted-foreground">
   469→                {windowedPoints.length === 0
   470→                  ? "No snapshots captured yet."
   471→                  : "1 snapshot captured — need at least 2 to draw a trend."}
   472→              </p>
   473→            </div>
   474→          </div>
   475→        ) : (
   476→          // --- CHART (2+ points) ---
   477→          <>
   478→            {/* Accessible label + visually-hidden table for screen readers.
   479→                The SVG below has role="img" with this concise aria-label.
   480→                The table carries the FULL series (every stored point in
   481→                the window) so screen-reader users get the same information
   482→                as sighted users — hover tooltips are NOT the accessible
   483→                path. */}
   484→            <div
   485→              role="img"
   486→              aria-label={`Compliance rate trend, last ${windowedPoints.length} snapshots across ${windowDays} days.`}
   487→            >
   488→              <ResponsiveContainer width="100%" height={240}>
   489→                <LineChart
   490→                  data={chartData}
   491→                  margin={{ left: 4, right: 16, top: 10, bottom: 0 }}
   492→                >
   493→                  <CartesianGrid
   494→                    strokeDasharray="3 3"
   495→                    vertical={false}
   496→                    stroke="hsl(var(--border))"
   497→                  />
   498→                  <XAxis
   499→                    dataKey="date"
   500→                    tickLine={false}
   501→                    axisLine={false}
   502→                    tick={{ fontSize: 11 }}
   503→                    // Abbreviated date format for the x-axis (compact).
   504→                    // Uses the shared date util — no hardcoded format.
   505→                    tickFormatter={(value: string) => {
   506→                      // Parse the YYYY-MM-DD as UTC (matches snapshot date semantics).
   507→                      const d = new Date(value + "T12:00:00Z");
   508→                      return formatAbsolute(d);
   509→                    }}
   510→                    // Show ~6 ticks to avoid crowding. Recharts auto-selects.
   511→                    minTickGap={30}
   512→                  />
   513→                  <YAxis
   514→                    domain={[0, 100]}
   515→                    ticks={[0, 25, 50, 75, 100]}
   516→                    tickLine={false}
   517→                    axisLine={false}
   518→                    tick={{ fontSize: 11 }}
   519→                    tickFormatter={(value: number) => `${value}%`}
   520→                    width={36}
   521→                  />
   522→                  <RechartsTooltip
   523→                    // Custom tooltip content — see TrendTooltip below.
   524→                    content={<TrendTooltip />}
   525→                    cursor={{ stroke: "hsl(var(--border))", strokeWidth: 1 }}
   526→                  />
   527→                  <Line
   528→                    type="monotone"
   529→                    dataKey="percentage"
   530→                    stroke="var(--status-compliant)"
   531→                    strokeWidth={2}
   532→                    dot={{ r: 2, fill: "var(--status-compliant)" }}
   533→                    activeDot={{ r: 4 }}
   534→                    name="Compliance rate"
   535→                    // connectNulls={false} is the DEFAULT — we make it
   536→                    // explicit so a future change to `true` doesn't
   537→                    // silently start interpolating across gaps.
   538→                    connectNulls={false}
   539→                    isAnimationActive={false}
   540→                  />
   541→                </LineChart>
   542→              </ResponsiveContainer>
   543→            </div>
   544→
   545→            {/* Visually-hidden table — the accessible path for screen
   546→                readers. Per 4.2b spec: "a visually-hidden data table
   547→                (date, counts, rate) carrying the full series for screen
   548→                readers." */}
   549→            <TrendAccessibleTable points={windowedPoints} />
   550→          </>
   551→        )}
   552→      </CardContent>
   553→    </Card>
   554→  );
   555→}

   557→// --- Custom tooltip: "12 Jun 2026 · 17 of 78 (22%)" — same phrasing as 1.2.
   558→function TrendTooltip({ active, payload }: {
   559→  active?: boolean;
   560→  payload?: Array<{ payload?: TrendChartDatum }>;
   561→}) {
   562→  if (!active || !payload || payload.length === 0) return null;
   563→  const datum = payload[0]?.payload;
   564→  if (!datum) return null;
   565→
   566→  // Gap rows render a "No data" tooltip — no fabricated number.
   567→  if (datum.isGap || !datum.point) {
   568→    return (
   569→      <div className="rounded-md border bg-background p-2 text-xs shadow-sm">
   570→        <p className="font-medium">{datum.date}</p>
   571→        <p className="text-muted-foreground">No data (no snapshot for this date)</p>
   572→      </div>
   573→    );
   574→  }
   575→
   576→  const p = datum.point;
   577→  // The percentage is pre-computed in the metrics module (1.2 invariant).
   578→  // We just render it. If null (zero denominator), show "—" instead of NaN.
   579→  const pctLabel = p.percentage === null ? "—" : `${p.percentage}%`;
   580→  return (
   581→    <div className="rounded-md border bg-background p-2 text-xs shadow-sm">
   582→      <p className="font-medium">
   583→        {/* "12 Jun 2026 · 17 of 78 (22%)" — matches 1.2 phrasing via formatAbsolute */}
   584→        {formatAbsolute(new Date(datum.date + "T12:00:00Z"))} · {p.rateNumerator} of {p.rateDenominator} ({pctLabel})
   585→      </p>
   586→      <p className="text-muted-foreground">
   587→        Compliant {p.compliant} · Expiring {p.expiring} · Overdue {p.overdue} · Missing {p.missing}
   588→      </p>
   589→    </div>
   590→  );
   591→}

   630→interface LollipopDatum {

   636→function LollipopChart({ data }: { data: LollipopDatum[] }) {
   637→  if (data.length === 0) return null;
   638→  const maxValue = Math.max(...data.map((d) => d.value));
   639→  const chartMax = Math.max(maxValue + 1, 5);
   640→  const STEM_AREA_WIDTH_PCT = 72;
   641→  return (
   642→    <div className="space-y-2.5">
   643→      {data.map((d, i) => {
   644→        const stemPct = (d.value / chartMax) * STEM_AREA_WIDTH_PCT;
   645→        return (
   646→          <div key={`${d.name}-${i}`} className="flex items-center gap-3">
   647→            <span className="shrink-0 truncate text-xs text-muted-foreground" style={{ width: "28%" }} title={d.name}>
   648→              {d.name}
   649→            </span>
   650→            <div className="relative flex-1" style={{ height: "16px" }}>
   651→              <div className="absolute top-1/2 -translate-y-1/2 rounded-full bg-muted" style={{ width: `${STEM_AREA_WIDTH_PCT}%`, height: "2px" }} />
   652→              <div className="absolute top-1/2 -translate-y-1/2 rounded-full" style={{ width: `${stemPct}%`, height: "2px", backgroundColor: d.fill, opacity: 0.5 }} />

   654→              <span className="absolute top-1/2 -translate-y-1/2 text-xs font-semibold" style={{ left: `calc(${stemPct}% + 12px)`, color: d.fill }}>

   656→              </span>

   658→          </div>

   688→function ComplianceCompositionBar({ counts, denominator, totalUnions, onFilter }: ComplianceCompositionBarProps) {
   689→  const tiles = getTileStatuses();
   690→
   691→  // Edge: zero denominator → "No requirements configured"
   692→  if (denominator === 0) {
   693→    return (
   694→      <Card>
   695→        <CardHeader>
   696→          <CardTitle className="text-lg">Document Compliance Composition</CardTitle>
   697→          <CardDescription>Compliance status breakdown across all unions.</CardDescription>
   698→        </CardHeader>
   699→        <CardContent>
   700→          <p className="text-sm text-muted-foreground">No requirements configured.</p>
   701→        </CardContent>
   702→      </Card>
   703→    );
   704→  }
   705→
   706→  // Denominator mismatch guard: sum of applicable counts must equal denominator.
   707→  // AWAITING_REVIEW is included in the applicable set per the rule in
   708→  // metrics.ts computeRateFromCounts (it's in the denominator).
   709→  const sum = counts.COMPLIANT + counts.EXPIRING + counts.OVERDUE + counts.AWAITING_REVIEW + counts.MISSING;
   710→  if (sum !== denominator) {
   711→    console.error(
   712→      `[ComplianceCompositionBar] Denominator mismatch: sum of counts (${sum}) != denominator (${denominator}). Not rendering.`,

   714→    return (
   715→      <Card>
   716→        <CardHeader>
   717→          <CardTitle className="text-lg">Document Compliance Composition</CardTitle>
   718→          <CardDescription>Compliance status breakdown across all unions.</CardDescription>
   719→        </CardHeader>
   720→        <CardContent>
   721→          <p className="text-sm text-destructive">Data inconsistency detected. Please refresh.</p>
   722→        </CardContent>
   723→      </Card>

   726→
   727→  // Segments: only non-zero counts render as visible bar segments.
   728→  // Zero-count statuses are omitted from the bar but still appear in the legend.
   729→  const segments = tiles
   730→    .map((status) => ({
   731→      status,
   732→      config: STATUS_CONFIG[status],
   733→      count: counts[status],
   734→      widthPct: (counts[status] / denominator) * 100,

   750→            <li key={status}>
   751→              {counts[status]} of {denominator} {STATUS_CONFIG[status].label}
   752→            </li>
   753→          ))}
   754→        </ul>
   755→
   756→        {/* Stacked bar: flex row of proportional segments */}
   757→        <TooltipProvider delayDuration={0}>
   758→          <div className="flex h-8 w-full overflow-hidden rounded-lg" role="group" aria-label="Compliance status composition">

   770→                      backgroundColor: seg.config.colorToken,

   772→                  >
   773→                    {/* Count inside segment: show only if the segment is wide enough
   774→                        (roughly > 8% of the bar — enough for a 2-digit number) */}
   775→                    {seg.widthPct > 8 && (
   776→                      <span className="text-xs font-semibold text-white mix-blend-difference">

   801→                <Tooltip>
   802→                  <TooltipTrigger asChild>
   803→                    <button
   804→                      type="button"
   805→                      aria-label={`${config.label}: ${count}`}
   806→                      onClick={() => onFilter?.(status)}
   807→                      disabled={!onFilter}

   816→                    </button>
