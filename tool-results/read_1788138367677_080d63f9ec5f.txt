   319→   319→function DocumentTile({
   320→   320→  doc,
   321→   321→  unionId,
   322→   322→  onUploadVersion,
   323→   323→}: {
   324→   324→  doc: UnionDocument;
   325→   325→  unionId: string;
   326→   326→  onUploadVersion: () => void;
   327→   327→}) {
   328→   328→  const [versionsOpen, setVersionsOpen] = useState(false);
   329→   329→  const [versions, setVersions] = useState<unknown[] | null>(null);
   330→   330→  const [versionsLoading, setVersionsLoading] = useState(false);
   331→   331→
   332→   332→  async function loadVersions() {
   333→   333→    setVersionsLoading(true);
   334→   334→    try {
   335→   335→      const res = await fetch(`/api/unions/${unionId}/documents/${doc.id}/versions`, {
   336→   336→        credentials: "include",
   337→   337→      });
   338→   338→      const data = await res.json();
   339→   339→      if (res.ok) {
   340→   340→        setVersions(data.versions);
   341→   341→      } else {
   342→   342→        toast.error("Failed to load versions.");
   343→   343→      }
   344→   344→    } catch {
   345→   345→      toast.error("Network error.");
   346→   346→    } finally {
   347→   347→      setVersionsLoading(false);
   348→   348→    }
   349→   349→  }
   350→   350→
   351→   351→  async function restoreVersion(versionId: string) {
   352→   352→    try {
   353→   353→      const res = await fetch(`/api/unions/${unionId}/documents/${doc.id}/restore`, {
   354→   354→        method: "POST",
   355→   355→        headers: { "Content-Type": "application/json" },
   356→   356→        credentials: "include",
   357→   357→        body: JSON.stringify({ versionId }),
   358→   358→      });
   359→   359→      const data = await res.json();
   360→   360→      if (!res.ok) {
   361→   361→        toast.error(data.error || "Restore failed.");
   362→   362→        return;
   363→   363→      }
   364→   364→      toast.success(`Restored from version ${data.restoredFromVersion}. New version: ${data.newVersion.versionNumber}.`);
   365→   365→      setVersionsOpen(false);
   366→   366→      // Reload the union list to reflect the new version
   367→   367→      window.location.reload();
   368→   368→    } catch {
   369→   369→      toast.error("Network error.");
   370→   370→    }
   371→   371→  }
   372→   372→
   373→   373→  return (
   374→   374→    <div className="rounded-md border p-3">
   375→   375→      <div className="flex items-start justify-between">
   376→   376→        <div className="flex items-start gap-2">
   377→   377→          <FileText className="mt-0.5 h-4 w-4 text-muted-foreground" />
   378→   378→          <div>
   379→   379→            <p className="text-sm font-medium">{doc.documentType}</p>
   380→   380→            {doc.disposedAt ? (
   381→   381→              <Badge variant="destructive" className="mt-1 text-xs">Disposed</Badge>
   382→   382→            ) : doc.latestVersion ? (
   383→   383→              <p className="mt-1 text-xs text-muted-foreground">
   384→   384→                v{doc.latestVersion.versionNumber} · {new Date(doc.latestVersion.uploadedAt).toLocaleDateString()}
   385→   385→              </p>
   386→   386→            ) : (
   387→   387→              <p className="mt-1 text-xs text-muted-foreground">No version uploaded</p>
   388→   388→            )}
   389→   389→          </div>
   390→   390→        </div>
   391→   391→      </div>
   392→   392→      <div className="mt-3 flex gap-1">
   393→   393→        <Button size="sm" variant="outline" className="h-7 text-xs" onClick={onUploadVersion}>
   394→   394→          <Upload className="mr-1 h-3 w-3" />
   395→   395→          New version
   396→   396→        </Button>
   397→   397→        {doc.latestVersion && !doc.disposedAt && (
   398→   398→          <a href={`/api/unions/${unionId}/documents/${doc.id}/download`} target="_blank" rel="noopener noreferrer">
   399→   399→            <Button size="sm" variant="ghost" className="h-7 text-xs">
   400→   400→              <Download className="mr-1 h-3 w-3" />
   401→   401→              Download
   402→   402→            </Button>
   403→   403→          </a>
   404→   404→        )}
   405→   405→        <Button
   406→   406→          size="sm"
   407→   407→          variant="ghost"
   408→   408→          className="h-7 text-xs"
   409→   409→          onClick={() => {
   410→   410→            if (!versionsOpen && !versions) loadVersions();
   411→   411→            setVersionsOpen(!versionsOpen);
   412→   412→          }}
   413→   413→        >
   414→   414→          <History className="mr-1 h-3 w-3" />
   415→   415→          History
   416→   416→        </Button>
   417→   417→      </div>
   418→   418→
   419→   419→      {versionsOpen && (
   420→   420→        <div className="mt-3 border-t pt-2">
   421→   421→          {versionsLoading ? (
   422→   422→            <p className="text-xs text-muted-foreground">Loading...</p>
   423→   423→          ) : versions && versions.length > 0 ? (
   424→   424→            <div className="space-y-1">
   425→   425→              {(versions as Array<{
   426→   426→                id: string;
   427→   427→                versionNumber: number;
   428→   428→                fileName: string;
   429→   429→                uploadedAt: string;
   430→   430→                isCurrent: boolean;
   431→   431→              }>).map((v) => (
   432→   432→                <div
   433→   433→                  key={v.id}
   434→   434→                  className="flex items-center justify-between rounded bg-muted/30 p-2 text-xs"
   435→   435→                >
   436→   436→                  <div>
   437→   437→                    <span className="font-medium">v{v.versionNumber}</span>
   438→   438→                    {v.isCurrent && <Badge className="ml-1 text-[10px]">Current</Badge>}
   439→   439→                    <p className="text-muted-foreground">{new Date(v.uploadedAt).toLocaleString()}</p>
   440→   440→                  </div>
   441→   441→                  {!v.isCurrent && (
   442→   442→                    <Button
   443→   443→                      size="sm"
   444→   444→                      variant="ghost"
   445→   445→                      className="h-6 text-xs"
   446→   446→                      onClick={() => restoreVersion(v.id)}
   447→   447→                    >
   448→   448→                      <RotateCcw className="mr-1 h-3 w-3" />
   449→   449→                      Restore
   450→   450→                    </Button>
   451→   451→                  )}
   452→   452→                </div>
   453→   453→              ))}
   454→   454→            </div>
   455→   455→          ) : (
   456→   456→            <p className="text-xs text-muted-foreground">No versions.</p>
   457→   457→          )}
   458→   458→        </div>
   459→   459→      )}
   460→   460→    </div>
   461→   461→  );
   462→   462→}

   468→   468→function EditUnionDialog({
   469→   469→  union,
   470→   470→  userRole,
   471→   471→  onClose,
   472→   472→  onSaved,
   473→   473→}: {
   474→   474→  union: Union;
   475→   475→  userRole: string;
   476→   476→  onClose: () => void;
   477→   477→  onSaved: () => void;
   478→   478→}) {
   479→   479→  const isRegistrar = userRole === "REGISTRAR";
   480→   480→  const isOfficer = userRole === "OFFICER";
   481→   481→
   482→   482→  const [unionName, setUnionName] = useState(union.unionName);

   673→   673→interface UploadDialogProps {
   674→   674→  unionId: string;
   675→   675→  defaultDocType: string;
   676→   676→  open: boolean;
   677→   677→  onClose: () => void;

   680→   680→
   681→   681→function UploadDialog({ unionId, defaultDocType, open, onClose, onUploaded }: UploadDialogProps) {
   682→   682→  const [documentType, setDocumentType] = useState(defaultDocType);
   683→   683→  const [title, setTitle] = useState("");
   684→   684→  const [file, setFile] = useState<File | null>(null);
   685→   685→  const [uploading, setUploading] = useState(false);
   686→   686→
   687→   687→  // Sync the doc type when the default changes (e.g., user clicks "New version" on a specific doc)
   688→   688→  useState(() => {
   689→   689→    if (defaultDocType) setDocumentType(defaultDocType);
   690→   690→  });
   691→   691→
   692→   692→  async function handleUpload(e: FormEvent) {
   693→   693→    e.preventDefault();
   694→   694→    if (!file) {
   695→   695→      toast.error("Please select a file.");
   696→   696→      return;
   697→   697→    }
   698→   698→    if (!documentType.trim()) {
   699→   699→      toast.error("Document type is required.");
   700→   700→      return;
   701→   701→    }
   702→   702→    if (!title.trim()) {
   703→   703→      toast.error("Title is required.");
   704→   704→      return;
   705→   705→    }
   706→   706→    setUploading(true);
   707→   707→    try {
   708→   708→      const formData = new FormData();
   709→   709→      formData.append("file", file);
   710→   710→      formData.append("documentType", documentType.trim());
   711→   711→      formData.append("title", title.trim());
   712→   712→      const res = await fetch(`/api/unions/${unionId}/documents`, {
   713→   713→        method: "POST",
   714→   714→        credentials: "include",
   715→   715→        body: formData,
   716→   716→      });
   717→   717→      const data = await res.json();
   718→   718→      if (!res.ok) {
   719→   719→        toast.error(data.error || "Upload failed.");
   720→   720→        return;
   721→   721→      }
   722→   722→      toast.success(`Uploaded ${documentType} v${data.version.versionNumber}.`);
   723→   723→      onUploaded();
   724→   724→    } catch {
   725→   725→      toast.error("Network error. Please try again.");
   726→   726→    } finally {
   727→   727→      setUploading(false);
   728→   728→    }
   729→   729→  }
   730→   730→
   731→   731→  return (
   732→   732→    <Dialog open={open} onOpenChange={(o) => !o && onClose()}>
   733→   733→      <DialogContent>
   734→   734→        <DialogHeader>
   735→   735→          <DialogTitle>Upload Document</DialogTitle>
   736→   736→          <DialogDescription>
   737→   737→            Upload a new version of a document. Duplicate detection is based on SHA-256 checksum.
   738→   738→          </DialogDescription>
   739→   739→        </DialogHeader>
   740→   740→        <form onSubmit={handleUpload} className="space-y-4">
   741→   741→          <div className="space-y-2">
   742→   742→            <Label htmlFor="docType">Document Type</Label>
   743→   743→            <Input
   744→   744→              id="docType"
   745→   745→              value={documentType}
   746→   746→              onChange={(e) => setDocumentType(e.target.value)}
   747→   747→              placeholder="e.g. Constitution, Financial Return"
   748→   748→            />
   749→   749→          </div>
   750→   750→          <div className="space-y-2">
   751→   751→            <Label htmlFor="title">Title</Label>
   752→   752→            <Input
   753→   753→              id="title"
   754→   754→              value={title}
   755→   755→              onChange={(e) => setTitle(e.target.value)}
   756→   756→              placeholder="A descriptive title for this version"
   757→   757→            />
   758→   758→          </div>
   759→   759→          <div className="space-y-2">
   760→   760→            <Label htmlFor="file">File</Label>
   761→   761→            <Input
   762→   762→              id="file"
   763→   763→              type="file"
   764→   764→              onChange={(e) => setFile(e.target.files?.[0] ?? null)}
   765→   765→              accept=".pdf,.doc,.docx,.jpg,.png"
   766→   766→            />
   767→   767→            {file && (
   768→   768→              <p className="text-xs text-muted-foreground">
   769→   769→                {file.name} ({(file.size / 1024).toFixed(1)} KB)
   770→   770→              </p>
   771→   771→            )}
   772→   772→          </div>
   773→   773→          <DialogFooter>
   774→   774→            <Button type="button" variant="outline" onClick={onClose} disabled={uploading}>
   775→   775→              Cancel
   776→   776→            </Button>
   777→   777→            <Button type="submit" disabled={uploading || !file}>
   778→   778→              {uploading ? (
   779→   779→                <>
   780→   780→                  <Loader2 className="mr-2 h-4 w-4 animate-spin" />
   781→   781→                  Uploading...
   782→   782→                </>
   783→   783→              ) : (
   784→   784→                <>
   785→   785→                  <Upload className="mr-2 h-4 w-4" />
   786→   786→                  Upload
   787→   787→                </>
   788→   788→              )}
   789→   789→            </Button>
   790→   790→          </DialogFooter>
   791→   791→        </form>
   792→   792→      </DialogContent>
   793→   793→    </Dialog>
   794→   794→  );
   795→   795→}
